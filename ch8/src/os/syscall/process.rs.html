36/// create a child process, which is a copy of the current one
37pub fn sys_fork() -> isize {
38    let current_process = current_process();
39    let new_task = current_process.fork();
40    let new_pid = new_task.getpid();
41    new_pid as isize
42}

44/// Maximum number of `argv` entries `sys_exec` will read. The user is supposed
45/// to NUL-terminate the pointer array; without a bound, a corrupted or
46/// out-of-bounds `argv` (e.g. the ch8 `usertests` harness historically did
47/// not NUL-terminate its 4-entry array) would make `translated_str` spin on an
48/// unterminated string.
49const MAX_ARGV: usize = 32;
50
51/// replace the current process with the program named by `path`, which is
52/// loaded from the file system; `args` is a NUL-terminated array of argument
53/// string pointers in the user address space.
54pub fn sys_exec(path: *const u8, mut args: *const usize) -> isize {
55    let token = current_user_token();
56    let path = translated_str(token, path);
57    let mut args_vec: Vec<String> = Vec::new();
58    loop {
59        let arg_str_ptr = *translated_ref(token, args);
60        if arg_str_ptr == 0 {
61            break;
62        }
63        args_vec.push(translated_str(token, arg_str_ptr as *const u8));
64        unsafe {
65            args = args.add(1);
66        }
67        if args_vec.len() >= MAX_ARGV {
68            warn!("exec: argv longer than {} entries, truncating", MAX_ARGV);
69            break;
70        }
71    }
72    if let Some(app_inode) = open_file(path.as_str(), OpenFlags::RDONLY) {
73        let all_data = app_inode.read_all();
74        let process = current_process();
75        let argc = args_vec.len();
76        process.exec(all_data.as_slice(), args_vec);
77        // return argc because cx.rdi will be covered with it later
78        argc as isize
79    } else {
80        -1
81    }
82}

84/// If there is not a child process whose pid is same as given, return -1.
85/// Else if there is a child process but it is still running, return -2.
86pub fn sys_waitpid(pid: isize, exit_code_ptr: *mut i32) -> isize {
87    let process = current_process();
88    // find a child process
89
90    // ---- access current TCB exclusively
91    let mut inner = process.inner_exclusive_access();
92    if !inner
93        .children
94        .iter()
95        .any(|p| pid == -1 || pid as usize == p.getpid())
96    {
97        return -1;
98        // ---- release current PCB
99    }
100    let pair = inner.children.iter().enumerate().find(|(_, p)| {
101        // ++++ temporarily access child PCB lock exclusively
102        p.inner_exclusive_access().is_zombie && (pid == -1 || pid as usize == p.getpid())
103        // ++++ release child PCB
104    });
105    if let Some((idx, _)) = pair {
106        let child = inner.children.remove(idx);
107        // SMP note: the child process is *not* deallocated here even though
108        // this may be the last reference held by the parent: the exiting
109        // main thread parks its own reference (with the still-live kernel
110        // stack) on the core's `exiting_process` slot and only releases it
111        // after it has switched away. So the kernel stack of the exited main
112        // thread is freed by that core's idle loop, never under the exiting
113        // thread itself. (single-core RISC-V asserted `strong_count == 1`
114        // here, which is not valid under SMP.)
115        let found_pid = child.getpid();
116        // ++++ temporarily access child TCB exclusively
117        let exit_code = child.inner_exclusive_access().exit_code;
118        // ++++ release child PCB
119        // x86-64: the trap handler runs on this task's page table, so the user
120        // pointer is directly dereferenceable
121        unsafe {
122            *exit_code_ptr = exit_code;
123        }
124        found_pid as isize
125    } else {
126        -2
127    }
128    // ---- release current PCB lock automatically
129}

131/// Send the signal `signum` to the process `pid`, if it is legal.
132pub fn sys_kill(pid: usize, signum: i32) -> isize {
133    if let Some(process) = pid2process(pid) {
134        if let Some(flag) = SignalFlags::from_bits(1 << signum) {
135            // insert the signal if legal
136            let mut process_ref = process.inner_exclusive_access();
137            if process_ref.signals.contains(flag) {
138                return -1;
139            }
140            process_ref.signals.insert(flag);
141            0
142        } else {
143            -1
144        }
145    } else {
146        -1
147    }
148}

150/// Change the signal mask of the current process, returning the old mask.
151pub fn sys_sigprocmask(mask: u32) -> isize {
152    let process = current_process();
153    let mut inner = process.inner_exclusive_access();
154    let old_mask = inner.signal_mask;
155    if let Some(flag) = SignalFlags::from_bits(mask) {
156        inner.signal_mask = flag;
157        old_mask.bits() as isize
158    } else {
159        -1
160    }
161}

163/// Return from a user signal handler: restore the trap context saved when the
164/// handler was entered.
165///
166/// x86-64 note: the syscall return value goes in `rax`, and after the trap
167/// context is restored the original user code resumes, so the value returned
168/// here is the one saved in `rax`.
169pub fn sys_sigreturn() -> isize {
170    let process = current_process();
171    let mut inner = process.inner_exclusive_access();
172    inner.handling_sig = -1;
173    // restore the trap context
174    let task = crate::task::current_task().unwrap();
175    let task_inner = task.inner_exclusive_access();
176    let trap_ctx = task_inner.get_trap_cx();
177    *trap_ctx = inner.trap_ctx_backup.unwrap();
178    // Here we return the value of rax in the trap_ctx,
179    // otherwise it will be overwritten after we trap
180    // back to the original execution of the application.
181    trap_ctx.rax as isize
182}

184fn check_sigaction_error(signal: SignalFlags, action: usize, old_action: usize) -> bool {
185    if action == 0
186        || old_action == 0
187        || signal == SignalFlags::SIGKILL
188        || signal == SignalFlags::SIGSTOP
189    {
190        true
191    } else {
192        false
193    }
194}

196/// Set the action of `signum`, returning the previous action through
197/// `old_action`.
198pub fn sys_sigaction(
199    signum: i32,
200    action: *const SignalAction,
201    old_action: *mut SignalAction,
202) -> isize {
203    let token = current_user_token();
204    let process = current_process();
205    let mut inner = process.inner_exclusive_access();
206    if signum as usize > MAX_SIG {
207        return -1;
208    }
209    if let Some(flag) = SignalFlags::from_bits(1 << signum) {
210        if check_sigaction_error(flag, action as usize, old_action as usize) {
211            return -1;
212        }
213        let prev_action = inner.signal_actions.table[signum as usize];
214        *translated_refmut(token, old_action) = prev_action;
215        inner.signal_actions.table[signum as usize] = *translated_ref(token, action);
216        0
217    } else {
218        -1
219    }
