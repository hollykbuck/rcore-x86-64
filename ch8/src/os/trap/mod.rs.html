1//! Trap handling functionality
2//!
3//! For rCore on x86-64, there are two entry points:
4//!
5//! - the `syscall` entry, whose address is stored in the `LSTAR` MSR. It
6//!   synthesizes an interrupt-like frame and forwards to [`trap_handler()`].
7//! - the exception stubs, pointed to by the IDT. The CPU switches to the
8//!   current task's kernel stack (the `rsp0` field of the TSS) and pushes the
9//!   interrupt frame, then each stub forwards to [`trap_handler()`].
10//!
11//! [`trap_handler()`] then calls different functionality based on what
12//! exactly the exception was. For example, syscalls go to [`syscall()`], and
13//! the APIC timer interrupt (vector 32) triggers task preemption.
14
15// Boot-time CPU structures are installed once in [`init()`] and then never
16// touched again, so `static mut` (via raw pointers) is fine here.
17#![allow(static_mut_refs)]
18// `trap.S` is written in GNU/AT&T syntax.
19#![allow(bad_asm_style)]

45lazy_static! {
46    /// the shared IDT (the handler addresses are identical on every core; each
47    /// core `lidt`s it separately)
48    static ref IDT: InterruptDescriptorTable = {
49        let mut idt = InterruptDescriptorTable::new();
50        for i in 0..256 {
51            idt.set_handler(i, unsafe { exception_stub_addrs[i] } as u64);
52        }
53        idt
54    };
55}

79/// Switch the kernel stack that the CPU uses for traps to `top`.
80///
81/// Both the `rsp0` field of this core's TSS (used for user-mode exceptions)
82/// and its `current_stack_top` (used by `syscall_entry`) are updated. The task
83/// manager calls this right before switching to another thread.
84pub fn set_current_stack_top(top: u64) {
85    let pc = current_per_cpu_mut();
86    pc.current_stack_top = top as usize;
87    pc.tss.set_rsp0(top);
88}

90/// The timer interrupt is enabled by the user RFLAGS IF bit (see
91/// [`context::TrapContext::app_init_context`]) together with the APIC timer
92/// setup done by [`crate::timer::init`]. The kernel itself deliberately keeps
93/// IF cleared (interrupt gates and `SFMASK`) so that external interrupts only
94/// happen in user mode, mirroring how RISC-V ch3 only enables `stimer` in
95/// user mode. Nothing needs to be done here.

98#[unsafe(no_mangle)]
99/// handle an interrupt, exception, or system call from user space
100pub fn trap_handler(cx: &mut TrapContext) -> &mut TrapContext {
101    match cx.trap_num {
102        TRAP_SYSCALL => {
103            let ret = syscall(cx.rax, [cx.rdi, cx.rsi, cx.rdx]);
104            // `cx` may have been replaced by `sys_exec` (it rebuilds the
105            // TrapContext on the kernel stack), so fetch it again
106            let cx = current_trap_cx();
107            cx.rax = ret as usize;
108        }
109        32 => {
110            // APIC timer interrupt (per-core local APIC)
111            timer::timer_eoi();
112            // only core 0 drives the global time base
113            if crate::cpu::current_cpu_id() == 0 {
114                timer::tick();
115            }
116            timer::set_next_trigger();
117            // wake up any threads whose sleep deadline has expired
118            timer::check_timer();
119            suspend_current_and_run_next();
120        }
121        14 => {
122            // page fault: CR2 holds the faulting linear address, error code
123            // bit 1 distinguishes a store (write) from a load (read) fault
124            let cr2: u64;
125            unsafe {
126                core::arch::asm!("mov rax, cr2", out("rax") cr2, options(nostack));
127            }
128            let access = if cx.error_code & 2 != 0 { "store" } else { "load" };
129            // diagnostic: identify the faulting thread and its user RSP
130            let (tid, rsp) = crate::task::current_task()
131                .map(|t| {
132                    let inner = t.inner_exclusive_access();
133                    (
134                        inner.res.as_ref().map(|r| r.tid).unwrap_or(usize::MAX),
135                        inner.get_trap_cx().rsp,
136                    )
137                })
138                .unwrap_or((usize::MAX, 0));
139            println!(
140                "[kernel] PageFault in application, {} addr = {:#x}, RIP = {:#x}, RSP = {:#x}, tid = {}, kernel killed it.",
141                access, cr2, cx.rip, rsp, tid
142            );
143            current_add_signal(SignalFlags::SIGSEGV);
144        }
145        13 | 6 => {
146            println!("[kernel] IllegalInstruction in application, kernel killed it.");
147            current_add_signal(SignalFlags::SIGILL);
148        }
149        _ => {
150            panic!(
151                "Unsupported trap: vector = {}, error_code = {:#x}, RIP = {:#x}",
152                cx.trap_num, cx.error_code, cx.rip
153            );
154        }
155    }
156    // handle the pending signals (including the ones just sent above)
157    handle_signals();
158    // the default action of an error signal is to kill the process
159    if let Some((errno, msg)) = check_signals_error_of_current() {
160        println!("[kernel] {}", msg);
161        exit_current_and_run_next(errno);
162    }
163    let cx = current_trap_cx();
164    cx
165}

167#[unsafe(no_mangle)]
168/// A trap that happened in kernel mode (i.e. a kernel bug): report and halt.
169pub extern "C" fn kernel_exception_panic(
170    vector: usize,
171    error_code: usize,
172    fault_rip: usize,
173    fault_cs: usize,
174) -> ! {
175    let cr2: u64;
176    let cr3: u64;
177    unsafe {
178        core::arch::asm!("mov rax, cr2", out("rax") cr2, options(nostack));
179        core::arch::asm!("mov rax, cr3", out("rax") cr3, options(nostack));
180    }
181    error!(
182        "[kernel] Exception #{} (error_code = {:#x}) occurred in kernel mode, CR2 = {:#x}, CR3 = {:#x}, RIP = {:#x}, CS = {:#x}, kernel halted.",
183        vector, error_code, cr2, cr3, fault_rip, fault_cs
184    );
185    crate::uart::shutdown(true)
186}

188#[unsafe(no_mangle)]
189/// The APIC timer interrupt while running in kernel mode (an idle core's
190/// `sti; hlt; cli` wakeup): EOI, advance the clock on core 0, and wake any
191/// threads whose sleep deadline has expired. Returns via `iretq` back to the
192/// idle loop.
193pub extern "C" fn kernel_timer_eoi() {
194    timer::timer_eoi();
195    if crate::cpu::current_cpu_id() == 0 {
196        timer::tick();
197    }
198    timer::check_timer();
199}
