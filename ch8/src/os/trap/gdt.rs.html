1//! The Global Descriptor Table
2//!
3//! Long mode ignores segment limits/base addresses, but the descriptors are
4//! still used to hold the privilege level (DPL) and the code/data flags, and
5//! the TSS is described here too.
6
7use core::arch::asm;
8
9/// kernel code segment selector (index 1, DPL 0)
10pub const KERNEL_CS: u16 = 0x08;
11/// kernel data segment selector (index 2, DPL 0)
12pub const KERNEL_DS: u16 = 0x10;
13/// TSS segment selector (index 5, DPL 0)
14pub const TSS_SELECTOR: u16 = 0x28;
15
16/// DPL of the user mode segments
17const DPL_USER: u64 = 3;
18const LIMIT_HIGH: u64 = 0xF << 48;
19const GRANULARITY_4K: u64 = 1 << 55;
20const LONG_MODE: u64 = 1 << 53;
21const DEFAULT_OP_SIZE: u64 = 1 << 54;
22
23/// Build a 64-bit code/data segment descriptor.
24///
25/// `access`: `0x9A` for a code segment, `0x92` for a data segment.
26const fn segment(access: u64, dpl: u64, long: bool, db: bool) -> u64 {
27    let mut upper = LIMIT_HIGH | GRANULARITY_4K;
28    if long {
29        upper |= LONG_MODE;
30    }
31    if db {
32        upper |= DEFAULT_OP_SIZE;
33    }
34    0xFFFF // limit low
35        | (access << 40)
36        | (dpl << 45)
37        | upper
38}
39
40/// Build the 64-bit TSS descriptor (two 8-byte entries).
41fn tss_segment(base: u64) -> (u64, u64) {
42    let limit = (core::mem::size_of::<super::tss::TaskStateSegment>() - 1) as u64;
43    let low = limit // limit low
44        | ((base & 0xFFFF) << 16) // base[15:0]
45        | (((base >> 16) & 0xFF) << 32) // base[23:16]
46        | (0x89 << 40) // present, available 64-bit TSS
47        | (((base >> 24) & 0xFF) << 56); // base[31:24]
48    let high = (base >> 32) & 0xFFFFFFFF; // base[63:32]
49    (low, high)
50}
51
52/// The GDT, with space for the TSS descriptor
53#[repr(C, align(8))]
54pub struct Gdt {
55    table: [u64; 8],
56}
57
58impl Gdt {
59    /// Create a GDT with the given `tss_base` in the TSS descriptor.
60    pub fn new(tss_base: u64) -> Self {
61        let (tss_low, tss_high) = tss_segment(tss_base);
62        Self {
63            table: [
64                0,                              // 0: null
65                segment(0x9A, 0, true, false),  // 1: kernel code
66                segment(0x92, 0, false, true),  // 2: kernel data
67                segment(0x9A, DPL_USER, true, false), // 3: user code
68                segment(0x92, DPL_USER, false, true), // 4: user data
69                tss_low,                        // 5: TSS (low)
70                tss_high,                       // 6: TSS (high)
71                0,
72            ],
73        }
74    }
75
76    /// Load the GDT and reload the segment registers, including `ltr`.
77    pub fn load(&'static self) {
78        #[repr(C, packed)]
79        struct GdtPointer {
80            limit: u16,
81            base: u64,
82        }
83        let ptr = GdtPointer {
84            limit: (self.table.len() * 8 - 1) as u16,
85            base: self.table.as_ptr() as u64,
86        };
87        unsafe {
88            // load the new GDT
89            asm!("lgdt [{ptr}]", ptr = in(reg) &ptr, options(readonly, nostack));
90            // reload CS via a far return
91            asm!(
92                "push {cs}",
93                "lea rax, [rip + 2f]",
94                "push rax",
95                "retfq",
96                "2:",
97                cs = const KERNEL_CS,
98                out("rax") _,
99                options(nostack)
100            );
101            // reload the data segments
102            asm!(
103                "mov ax, {ds}",
104                "mov ds, ax",
105                "mov es, ax",
106                "mov fs, ax",
107                "mov gs, ax",
108                "mov ss, ax",
109                ds = const KERNEL_DS,
110                out("ax") _,
111                options(nostack)
112            );
113            // load the TSS
114            asm!(
115                "mov ax, {sel}",
116                "ltr ax",
117                sel = const TSS_SELECTOR,
118                out("ax") _,
119                options(nostack)
120            );
121        }
122    }
123}
