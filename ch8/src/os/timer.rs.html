1//! Timer setup (local APIC timer for both preemption and timekeeping)
2//!
3//! Limine masks both the 8259 PICs and the IO-APIC before handing over, and
4//! the legacy PIT path is unreliable under QEMU's TCG emulation (interrupts
5//! are easily lost when the handler is slow). The local APIC timer, on the
6//! other hand, fires as reliably as the RISC-V CLINT timer, so it is used for
7//! both the time-slice rotation (round-robin preemption) and as the time
8//! source.
9//!
10//! The preemption rate is set to 1000 Hz. QEMU's TCG only advances its
11//! virtual clock in real time or under `-icount`, so `-icount shift=auto` is
12//! recommended to make the time slices evenly instruction-budgeted rather
13//! than host-load dependent (see HANDOVER §5.5).
14//!
15//! Timekeeping reconstructs a free-running monotonic counter from the APIC
16//! timer, the x86-64 analogue of the RISC-V `mtime`:
17//!
18//! - In periodic mode the current-count register (FEE0 0390H / x2APIC MSR
19//!   0x839) is automatically reloaded from the initial-count register each
20//!   time the count reaches zero (Intel SDM Vol. 3A §13.5.4).
21//! - So `elapsed_ticks = wraps * INITIAL_COUNT + (INITIAL_COUNT - current)`
22//!   is a free-running monotonic counter at `APIC_TIMER_FREQ` ticks/second.
23//! - The kernel runs with IF=0, so the timer interrupt cannot preempt the two
24//!   reads inside `elapsed_ticks()`: `wraps` and `current` are always
25//!   mutually consistent, and the counter is inherently monotonic.
26//!
27//! The APIC timer input frequency is exactly 1 GHz on QEMU (measured: a count
28//! of 10,000,000 gives 100 Hz). On real hardware it is the bus/crystal clock
29//! (SDM §13.5.4), so the hardcoded 1 GHz only holds for QEMU -- fine for this
30//! tutorial. No RTC is involved anymore: the counter is guest-side and
31//! inherently monotonic, so the previous RTC + minute-wrap bookkeeping is
32//! gone.
33

45/// number of timer interrupts per second (the preemption rate)
46const TICKS_PER_SEC: usize = 1000;
47/// the APIC timer input frequency, measured on QEMU (1 GHz)
48const APIC_TIMER_FREQ: usize = 1_000_000_000;
49/// the count programmed into the APIC timer (period = INITIAL_COUNT ns)
50const INITIAL_COUNT: usize = APIC_TIMER_FREQ / TICKS_PER_SEC;
51
52/// the MSR holding the local APIC base/enable state
53const APIC_BASE_MSR: u32 = 0x1B;
54/// x2APIC enable bit in the APIC base MSR
55const APIC_BASE_X2: u64 = 1 << 10;
56
57/// LAPIC register offsets
58const LAPIC_EOI: u32 = 0x0B0;
59const LAPIC_LVT_TIMER: u32 = 0x320;
60const LAPIC_TIMER_INIT_COUNT: u32 = 0x380;
61#[allow(unused)]
62const LAPIC_TIMER_CURRENT_COUNT: u32 = 0x390;
63const LAPIC_TIMER_DIVIDE: u32 = 0x3E0;
64
65/// number of completed timer periods (interrupts delivered) on the bootstrap
66/// processor since boot. This is the global time base (1 tick = 1 ms at
67/// `TICKS_PER_SEC` Hz). Only core 0's timer interrupt increments it, so it is
68/// independent of the core on which `get_time_ms` happens to run.
69static GLOBAL_TICKS: AtomicUsize = AtomicUsize::new(0);
70
71unsafe fn rdmsr(msr: u32) -> u64 {
72    let mut hi: u32;
73    let mut lo: u32;
74    unsafe {
75        asm!(
76            "rdmsr",
77            in("ecx") msr,
78            out("eax") lo,
79            out("edx") hi,
80            options(nostack)
81        );
82    }
83    ((hi as u64) << 32) | lo as u64
84}
85
86unsafe fn wrmsr(msr: u32, value: u64) {
87    unsafe {
88        asm!(
89            "wrmsr",
90            in("ecx") msr,
91            in("eax") value as u32,
92            in("edx") (value >> 32) as u32,
93            options(nostack)
94        );
95    }
96}

98/// Write a local APIC register. Works in both xAPIC (MMIO) and x2APIC (MSR)
99/// modes.
100fn lapic_write(offset: u32, value: u32) {
101    unsafe {
102        let apic_base = rdmsr(APIC_BASE_MSR);
103        if apic_base & APIC_BASE_X2 != 0 {
104            wrmsr(0x800 + offset, value as u64);
105        } else {
106            // xAPIC MMIO: reach the physical window through the physmap
107            let base = crate::mm::phys_to_virt((apic_base & !0xFFF) as usize) as *mut u32;
108            core::ptr::write_volatile(base.add((offset / 4) as usize), value);
109        }
110    }
111}
112
113/// Read a local APIC register. Works in both xAPIC (MMIO) and x2APIC (MSR)
114/// modes.
115#[allow(unused)]
116fn lapic_read(offset: u32) -> u32 {
117    unsafe {
118        let apic_base = rdmsr(APIC_BASE_MSR);
119        if apic_base & APIC_BASE_X2 != 0 {
120            rdmsr(0x800 + offset) as u32
121        } else {
122            let base = crate::mm::phys_to_virt((apic_base & !0xFFF) as usize) as *const u32;
123            core::ptr::read_volatile(base.add((offset / 4) as usize))
124        }
125    }
126}

128/// Program the current core's APIC timer in periodic mode to interrupt at
129/// `TICKS_PER_SEC` Hz on vector 32. Called by every processor.
130pub fn init() {
131    // divide by 1
132    lapic_write(LAPIC_TIMER_DIVIDE, 0x0B);
133    // periodic mode, vector 32, unmasked
134    lapic_write(LAPIC_LVT_TIMER, 0x20 | (1 << 17));
135    // count for TICKS_PER_SEC Hz
136    lapic_write(LAPIC_TIMER_INIT_COUNT, INITIAL_COUNT as u32);
137    info!("timer: APIC timer interrupt at ~{} Hz", TICKS_PER_SEC);
138}

140/// Count one timer interrupt (one full period elapsed). Called from the timer
141/// interrupt handler on the **bootstrap processor only**; this drives the
142/// global time base.
143pub fn tick() {
144    GLOBAL_TICKS.fetch_add(1, AtomicOrdering::Relaxed);
145}
146
147/// Send the End-Of-Interrupt to the local APIC. Must be called at the start
148/// of the timer handler, otherwise no further interrupt is delivered.
149pub fn timer_eoi() {
150    lapic_write(LAPIC_EOI, 0);
151}

153/// get current time in milliseconds
154///
155/// Derived from the number of periods completed by the bootstrap processor's
156/// APIC timer, so the resolution is 1 ms (and monotonic by construction,
157/// independent of which core calls it).
158pub fn get_time_ms() -> usize {
159    GLOBAL_TICKS.load(AtomicOrdering::Relaxed)
160}

162/// set the next timer interrupt
163///
164/// The APIC timer is programmed in periodic mode, so it reloads its count
165/// automatically and there is nothing to reprogram per tick. Kept for
166/// symmetry with the RISC-V port.
167pub fn set_next_trigger() {}

169/// A timer event: wake `task` once `expire_ms` is reached.
170pub struct TimerCondVar {
171    pub expire_ms: usize,
172    pub task: Arc<TaskControlBlock>,
173}
174
175impl PartialEq for TimerCondVar {
176    fn eq(&self, other: &Self) -> bool {
177        self.expire_ms == other.expire_ms
178    }
179}
180impl Eq for TimerCondVar {}
181impl PartialOrd for TimerCondVar {
182    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
183        let a = -(self.expire_ms as isize);
184        let b = -(other.expire_ms as isize);
185        Some(a.cmp(&b))
186    }
187}
188
189impl Ord for TimerCondVar {
190    fn cmp(&self, other: &Self) -> Ordering {
191        self.partial_cmp(other).unwrap()
192    }
193}
194
195lazy_static! {
196    /// a min-heap of pending timer events
197    static ref TIMERS: UPSafeCell<BinaryHeap<TimerCondVar>> =
198        unsafe { UPSafeCell::new(BinaryHeap::<TimerCondVar>::new()) };
199}
200
201/// Add a timer event that wakes `task` at `expire_ms`.
202pub fn add_timer(expire_ms: usize, task: Arc<TaskControlBlock>) {
203    let mut timers = TIMERS.exclusive_access();
204    timers.push(TimerCondVar { expire_ms, task });
205}
206
207/// Remove all timer events of `task` (used when a process terminates).
208pub fn remove_timer(task: Arc<TaskControlBlock>) {
209    let mut timers = TIMERS.exclusive_access();
210    let mut temp = BinaryHeap::<TimerCondVar>::new();
211    for condvar in timers.drain() {
212        if Arc::as_ptr(&task) != Arc::as_ptr(&condvar.task) {
213            temp.push(condvar);
214        }
215    }
216    timers.clear();
217    timers.append(&mut temp);
218}
219
220/// Wake up every task whose timer has expired. Called from the timer
221/// interrupt handler.
222pub fn check_timer() {
223    let current_ms = get_time_ms();
224    let mut timers = TIMERS.exclusive_access();
225    while let Some(timer) = timers.peek() {
226        if timer.expire_ms <= current_ms {
227            wakeup_task(Arc::clone(&timer.task));
228            timers.pop();
229        } else {
230            break;
231        }
232    }
233}
