1//! Task management implementation
2//!
3//! Everything about task management, like starting and switching tasks is
4//! implemented here.
5//!
6//! A single global instance of [`TaskManager`] called `TASK_MANAGER` controls
7//! all the threads in the whole operating system.
8//!
9//! A single global instance of [`Processor`] called `PROCESSOR` monitors the
10//! running thread(s) for each core.
11//!
12//! A single global instance of [`PidAllocator`] called `PID_ALLOCATOR`
13//! allocates pids for user apps.
14//!
15//! Be careful when you see `__switch` ASM function in `switch.S`. Control flow around this function
16//! might not be what you expect.

55/// The pid of the init process

58/// Suspend the current 'Running' thread and run the next thread in the ready queue.
59pub fn suspend_current_and_run_next() {
60    // There must be an application running.
61    let task = take_current_task().unwrap();
62
63    // ---- access current TCB exclusively
64    let mut task_inner = task.inner_exclusive_access();
65    let task_cx_ptr = &mut task_inner.task_cx as *mut TaskContext;
66    // Change status to Ready
67    task_inner.task_status = TaskStatus::Ready;
68    drop(task_inner);
69    // ---- release current TCB
70
71    // SMP: the thread must not re-enter the global ready queue until its
72    // context has been saved by `__switch` below, otherwise another core could
73    // fetch it first and resume stale state. Park it in this core's pending
74    // queue; the idle loop drains it after the switch.
75    crate::cpu::current_per_cpu_mut()
76        .processor
77        .pending_requeue
78        .exclusive_access()
79        .push_back(task);
80    // jump to the scheduling cycle
81    schedule(task_cx_ptr);
82}

84/// Block the current 'Running' thread and run the next thread in the ready queue.
85pub fn block_current_and_run_next() {
86    let task = take_current_task().unwrap();
87    let mut task_inner = task.inner_exclusive_access();
88    let task_cx_ptr = &mut task_inner.task_cx as *mut TaskContext;
89    task_inner.task_status = TaskStatus::Blocked;
90    drop(task_inner);
91    schedule(task_cx_ptr);
92}

94/// Exit the current 'Running' thread and run the next thread.
95///
96/// The exiting thread becomes a zombie (its `exit_code` is recorded, its user
97/// resources are freed); the process terminates at once if this was its main
98/// thread.
99pub fn exit_current_and_run_next(exit_code: i32) {
100    let task = take_current_task().unwrap();
101    let mut task_inner = task.inner_exclusive_access();
102    let process = task.process.upgrade().unwrap();
103    let tid = task_inner.res.as_ref().unwrap().tid;
104    // record exit code
105    task_inner.exit_code = Some(exit_code);
106    task_inner.res = None;
107    // here we do not remove the thread since we are still using the kstack
108    // it will be deallocated when sys_waittid is called
109    drop(task_inner);
110    drop(task);
111    // however, if this is the main thread of current process
112    // the process should terminate at once
113    if tid == 0 {
114        let pid = process.getpid();
115        if pid == IDLE_PID {
116            println!(
117                "[kernel] Idle process exit with exit_code {} ...",
118                exit_code
119            );
120            if exit_code != 0 {
121                shutdown(true)
122            } else {
123                shutdown(false)
124            }
125        }
126        remove_from_pid2process(pid);
127        let mut process_inner = process.inner_exclusive_access();
128        // mark this process as a zombie process
129        process_inner.is_zombie = true;
130        // record exit code of main process
131        process_inner.exit_code = exit_code;
132
133        // Detach the child processes and reparent them to the init process.
134        // The reparenting itself only needs our own lock (the parent field of
135        // the orphans). They are *adopted* by `INITPROC` later, without our
136        // own lock held. On SMP this matters: otherwise the exit path (holds
137        // self, wants INITPROC), the parent's `waitpid` (holds parent, wants
138        // self) and the init process's `waitpid` (holds INITPROC, wants
139        // parent) form a lock cycle `self -> INITPROC -> parent -> self` that
140        // deadlocks whenever the three interleave.
141        let mut orphans: Vec<Arc<ProcessControlBlock>> = Vec::new();
142        for child in process_inner.children.drain(..) {
143            child.inner_exclusive_access().parent = Some(Arc::downgrade(&INITPROC));
144            orphans.push(child);
145        }
146
147        // deallocate user res (including tid/ustack) of all threads
148        // it has to be done before we dealloc the whole memory_set
149        // otherwise they will be deallocated twice
150        let mut recycle_res = Vec::<TaskUserRes>::new();
151        for task in process_inner.tasks.iter().filter(|t| t.is_some()) {
152            let task = task.as_ref().unwrap();
153            // if other tasks are Ready in TaskManager or waiting for a timer
154            // to be expired, we should remove them.
155            remove_inactive_task(Arc::clone(&task));
156            let mut task_inner = task.inner_exclusive_access();
157            if let Some(res) = task_inner.res.take() {
158                recycle_res.push(res);
159            }
160        }
161        // dealloc_tid and dealloc_user_res require access to PCB inner, so we
162        // need to collect those user res first, then release process_inner
163        // for now to avoid deadlock/double borrow problem.
164        drop(process_inner);
165
166        // adopt the orphaned children under the init process (we no longer
167        // hold our own lock, so no lock cycle with the waiting parents)
168        {
169            let mut initproc_inner = INITPROC.inner_exclusive_access();
170            for orphan in orphans.into_iter() {
171                initproc_inner.children.push(orphan);
172            }
173        }
174
175        recycle_res.clear();
176
177        let mut process_inner = process.inner_exclusive_access();
178        // deallocate other data in user space i.e. program code/data section
179        process_inner.memory_set.recycle_data_pages();
180        // drop file descriptors
181        process_inner.fd_table.clear();
182        // Remove all tasks except for the main thread itself.
183        // This is because we are still using the kstack under the TCB
184        // of the main thread. This TCB, including its kstack, will be
185        // deallocated when the process is reaped via waitpid.
186        while process_inner.tasks.len() > 1 {
187            process_inner.tasks.pop();
188        }
189        drop(process_inner);
190        // Park the process on this core: the final drop (which frees the main
191        // thread's kernel stack) must only happen after `__switch` below has
192        // switched away from this very stack. The idle loop releases it.
193        crate::cpu::current_per_cpu_mut().processor.exiting_process = Some(process);
194        // we do not have to save task context
195        let mut _unused = TaskContext::zero_init();
196        schedule(&mut _unused as *mut _);
197        unreachable!();
198    }
199    drop(process);
200    // we do not have to save task context
201    let mut _unused = TaskContext::zero_init();
202    schedule(&mut _unused as *mut _);
203    unreachable!();
204}

206/// Remove a thread from the ready queue and the timer queue (used when the
207/// process terminates while some of its threads are still inactive).
208pub fn remove_inactive_task(task: Arc<TaskControlBlock>) {
209    remove_task(Arc::clone(&task));
210    crate::timer::remove_timer(Arc::clone(&task));
211    // SMP: the thread may be parked in some core's pending queue, waiting to
212    // re-enter the ready queue after its context save; remove it there too.
213    crate::task::processor::remove_from_all_pending(task);
214}

230/// If the current process has a signal whose default action is to terminate
231/// it, return the (exit code, message) pair so the trap handler can kill it.
232pub fn check_signals_error_of_current() -> Option<(i32, &'static str)> {
233    let process = current_process();
234    let process_inner = process.inner_exclusive_access();

238/// Add a signal to the current process's pending set.
239pub fn current_add_signal(signal: SignalFlags) {
240    let process = current_process();
241    let mut process_inner = process.inner_exclusive_access();
242    process_inner.signals |= signal;

245/// The kernel's own handling of the "kernel" signals: SIGSTOP freezes the
246/// process, SIGCONT unfreezes it, everything else (SIGKILL/SIGDEF) kills it.
247fn call_kernel_signal_handler(signal: SignalFlags) {
248    let process = current_process();
249    let mut process_inner = process.inner_exclusive_access();
250    match signal {
251        SignalFlags::SIGSTOP => {
252            process_inner.frozen = true;
253            process_inner.signals ^= SignalFlags::SIGSTOP;
254        }
255        SignalFlags::SIGCONT => {
256            if process_inner.signals.contains(SignalFlags::SIGCONT) {
257                process_inner.signals ^= SignalFlags::SIGCONT;
258                process_inner.frozen = false;
259            }
260        }
261        _ => {
262            process_inner.killed = true;
263        }
264    }
265}

267/// Dispatch `signal` to the user's registered handler on the current thread.
268///
269/// x86-64 note: the trap context's `rip` is overwritten with the handler
270/// address (the `iretq` pops `RIP` from the frame) and the signal number is
271/// passed as the first SysV argument `rdi`. `sigreturn` restores the backup.
272fn call_user_signal_handler(sig: usize, signal: SignalFlags) {
273    let process = current_process();
274    let mut process_inner = process.inner_exclusive_access();
275
276    let handler = process_inner.signal_actions.table[sig].handler;
277    if handler != 0 {
278        // user handler
279        process_inner.handling_sig = sig as isize;
280        process_inner.signals ^= signal;
281
282        // backup the current trap context
283        let task = current_task().unwrap();
284        let task_inner = task.inner_exclusive_access();
285        let trap_ctx = task_inner.get_trap_cx();
286        process_inner.trap_ctx_backup = Some(*trap_ctx);
287
288        // jump to the handler on return to user mode
289        trap_ctx.rip = handler;
290        // pass the signal number as the first argument
291        trap_ctx.rdi = sig;
292    } else {
293        println!("[K] task/call_user_signal_handler: default action: ignore it or kill process");
294    }
295}

297fn check_pending_signals() {
298    for sig in 0..(MAX_SIG + 1) {
299        let process = current_process();
300        let process_inner = process.inner_exclusive_access();
301        let signal = SignalFlags::from_bits(1 << sig).unwrap();
302        if process_inner.signals.contains(signal) && (!process_inner.signal_mask.contains(signal))
303        {
304            let mut masked = true;
305            let handling_sig = process_inner.handling_sig;
306            if handling_sig == -1 {
307                masked = false;
308            } else {
309                let handling_sig = handling_sig as usize;
310                if !process_inner.signal_actions.table[handling_sig]
311                    .mask
312                    .contains(signal)
313                {
314                    masked = false;
315                }
316            }
317            if !masked {
318                drop(process_inner);
319                drop(process);
320                if signal == SignalFlags::SIGKILL
321                    || signal == SignalFlags::SIGSTOP
322                    || signal == SignalFlags::SIGCONT
323                    || signal == SignalFlags::SIGDEF
324                {
325                    // signal is a kernel signal
326                    call_kernel_signal_handler(signal);
327                } else {
328                    // signal is a user signal: run the handler and return
329                    call_user_signal_handler(sig, signal);
330                    return;
331                }
332            }
333        }
334    }
335}

337/// Called at the end of the trap handler: run any pending signal handlers. A
338/// process frozen by SIGSTOP stays suspended until SIGCONT wakes it up.
339pub fn handle_signals() {
340    loop {
341        check_pending_signals();
342        let (frozen, killed) = {
343            let process = current_process();
344            let process_inner = process.inner_exclusive_access();
345            (process_inner.frozen, process_inner.killed)
346        };
347        if !frozen || killed {
348            break;
349        }
350        suspend_current_and_run_next();
351    }
