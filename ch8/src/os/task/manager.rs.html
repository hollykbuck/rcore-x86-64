1//! Implementation of [`TaskManager`] and the pid -> process table.
2use super::{ProcessControlBlock, TaskControlBlock, TaskStatus};
3use crate::sync::UPSafeCell;
4use alloc::collections::{BTreeMap, VecDeque};
5use alloc::sync::Arc;
6use lazy_static::*;
7
8/// A ready queue of `TaskControlBlock`s.
9pub struct TaskManager {
10    ready_queue: VecDeque<Arc<TaskControlBlock>>,
11}
12
13/// A simple FIFO scheduler.
14impl TaskManager {
15    /// Create an empty `TaskManager`.
16    pub fn new() -> Self {
17        Self {
18            ready_queue: VecDeque::new(),
19        }
20    }
21    /// Add a thread to the ready queue.
22    pub fn add(&mut self, task: Arc<TaskControlBlock>) {
23        self.ready_queue.push_back(task);
24    }
25    /// Remove a thread from the ready queue.
26    pub fn remove(&mut self, task: Arc<TaskControlBlock>) {
27        if let Some((id, _)) = self
28            .ready_queue
29            .iter()
30            .enumerate()
31            .find(|(_, t)| Arc::as_ptr(t) == Arc::as_ptr(&task))
32        {
33            self.ready_queue.remove(id);
34        }
35    }
36    /// Remove the first thread and return it, or `None` if the queue is empty.
37    pub fn fetch(&mut self) -> Option<Arc<TaskControlBlock>> {
38        self.ready_queue.pop_front()
39    }
40}
41
42lazy_static! {
43    pub static ref TASK_MANAGER: UPSafeCell<TaskManager> =
44        unsafe { UPSafeCell::new(TaskManager::new()) };
45    /// a pid -> process map, used to send signals and reap a specific process
46    pub static ref PID2PROCESS: UPSafeCell<BTreeMap<usize, Arc<ProcessControlBlock>>> =
47        unsafe { UPSafeCell::new(BTreeMap::new()) };
48}
49
50/// Interface offered to add a thread.
51pub fn add_task(task: Arc<TaskControlBlock>) {
52    TASK_MANAGER.exclusive_access().add(task);
53}
54
55/// Wake up a blocked thread: mark it ready and put it back in the queue.
56///
57/// SMP: the thread is not added to the global ready queue directly. It is
58/// parked on the pending queue of its `last_cpu` (the core that owns its
59/// context save); that core's idle loop moves it to the ready queue once the
60/// context has been saved by `__switch`. This closes the race where a woken
61/// thread is resumed by another core before its block path finished saving
62/// its context.
63pub fn wakeup_task(task: Arc<TaskControlBlock>) {
64    let mut task_inner = task.inner_exclusive_access();
65    task_inner.task_status = TaskStatus::Ready;
66    drop(task_inner);
67    let owner = task.last_cpu.load(core::sync::atomic::Ordering::Relaxed);
68    super::processor::park_pending(owner, task);
69}
70
71/// Remove a thread from the ready queue (used when a process terminates).
72pub fn remove_task(task: Arc<TaskControlBlock>) {
73    TASK_MANAGER.exclusive_access().remove(task);
74}
75
76/// Interface offered to pop the first thread.
77pub fn fetch_task() -> Option<Arc<TaskControlBlock>> {
78    TASK_MANAGER.exclusive_access().fetch()
79}

81/// Look up a process by pid.
82pub fn pid2process(pid: usize) -> Option<Arc<ProcessControlBlock>> {
83    let map = PID2PROCESS.exclusive_access();
84    map.get(&pid).map(Arc::clone)
85}
86
87/// Insert a process into the pid -> process map.
88pub fn insert_into_pid2process(pid: usize, process: Arc<ProcessControlBlock>) {
89    PID2PROCESS.exclusive_access().insert(pid, process);
90}
91
92/// Remove a process from the pid -> process map (when it is reaped).
93pub fn remove_from_pid2process(pid: usize) {
94    let mut map = PID2PROCESS.exclusive_access();
95    if map.remove(&pid).is_none() {
96        panic!("cannot find pid {} in pid2process!", pid);
97    }
