1//! VirtIO block device driver (VirtIO over PCI, `virtio-drivers` 0.2.0).
2//!
3//! The `Hal` implementation uses the kernel's physmap (`pa + hhdm_offset`)
4//! for `phys_to_virt` and the kernel page table for DMA address translation.
5

7    FrameTracker, PhysAddr, PhysPageNum, StepByOne, VirtAddr, frame_alloc, frame_dealloc,

20pub struct VirtioHal;
21
22lazy_static! {
23    static ref QUEUE_FRAMES: UPSafeCell<Vec<FrameTracker>> =
24        unsafe { UPSafeCell::new(Vec::new()) };
25}
26
27impl Hal for VirtioHal {
28    fn dma_alloc(pages: usize) -> usize {
29        let mut ppn_base = PhysPageNum(0);
30        for i in 0..pages {
31            let frame = frame_alloc().unwrap();
32            if i == 0 {
33                ppn_base = frame.ppn;
34            }
35            // DMA needs contiguous physical pages; the stack allocator hands
36            // them out in order, so this holds in practice (same as the
37            // RISC-V tutorial).
38            assert_eq!(frame.ppn.0, ppn_base.0 + i);
39            QUEUE_FRAMES.exclusive_access().push(frame);
40        }
41        let pa: PhysAddr = ppn_base.into();
42        pa.0
43    }
44    fn dma_dealloc(pa: usize, pages: usize) -> i32 {
45        let pa = PhysAddr::from(pa);
46        let mut ppn_base: PhysPageNum = pa.into();
47        for _ in 0..pages {
48            frame_dealloc(ppn_base);
49            ppn_base.step();
50        }
51        0
52    }
53    fn phys_to_virt(paddr: usize) -> usize {
54        phys_to_virt(paddr)
55    }
56    fn share(buffer: NonNull<[u8]>, _direction: BufferDirection) -> usize {
57        let vaddr = buffer.as_ptr() as *mut u8 as usize;
58        crate::mm::PageTable::from_token(kernel_token())
59            .translate_va(VirtAddr::from(vaddr))
60            .unwrap()
61            .0
62    }
63    fn unshare(_paddr: usize, _buffer: NonNull<[u8]>, _direction: BufferDirection) {}
64}

66pub struct VirtIOBlock(UPSafeCell<VirtIOBlk<VirtioHal, PciTransport>>);
67
68// All access to the device goes through the `UPSafeCell` in a single-threaded
69// kernel, so we can safely claim Send/Sync even though the raw pointers inside
70// `virtio-drivers`' PciTransport are not.
71unsafe impl Send for VirtIOBlock {}
72unsafe impl Sync for VirtIOBlock {}
73
74impl VirtIOBlock {
75    pub fn new() -> Self {
76        let mut pci = crate::drivers::pci::Pci::new();
77        let bdf = pci
78            .find_block()
79            .expect("no VirtIO block device found on the PCI bus");
80        let transport =
81            PciTransport::new::<VirtioHal>(pci.root(), bdf).expect("failed to init PCI transport");
82        let blk = VirtIOBlk::new(transport).expect("failed to init VirtIO block device");
83        info!("virtio-blk capacity: {} sectors", blk.capacity());
84        Self(unsafe { UPSafeCell::new(blk) })
85    }
86}
87
88impl BlockDevice for VirtIOBlock {
89    fn read_block(&self, block_id: usize, buf: &mut [u8]) {
90        self.0
91            .exclusive_access()
92            .read_block(block_id, buf)
93            .expect("Error when reading VirtIOBlk");
94    }
95    fn write_block(&self, block_id: usize, buf: &[u8]) {
96        self.0
97            .exclusive_access()
98            .write_block(block_id, buf)
99            .expect("Error when writing VirtIOBlk");
100    }
