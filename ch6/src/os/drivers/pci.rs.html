1//! PCI bus enumeration for QEMU `q35`.
2//!
3//! The `q35` machine exposes the PCIe Enhanced Configuration Access Mechanism
4//! (ECAM) as MMIO at a base programmed in the MCH's `PCIEXBAR` register
5//! (config space offset `0x60`). OVMF may move it away from QEMU's default
6//! `0xB0000000`, so we read it at runtime through the legacy IO-port CAM
7//! (`0xCF8`/`0xCFC`), map the window through the physmap, and then use
8//! `virtio-drivers`' `PciRoot`/`PciTransport` over that MMIO ECAM.
9
10use crate::mm::ioremap;
11use core::arch::asm;
12use log::*;
13use virtio_drivers::transport::pci::bus::{BarInfo, Cam, DeviceFunction, MemoryBarType, PciRoot};
14use virtio_drivers::transport::pci::virtio_device_type;
15use virtio_drivers::transport::DeviceType;
16
17/// Bus 0 of the ECAM: 32 devices * 8 functions * 256 B configuration space
18/// = 1 MiB per bus.
19const ECAM_BUS0_SIZE: usize = 0x10_0000;
20
21/// Read a 32-bit word from PCI configuration space through the legacy IO
22/// port CAM (works before the ECAM is mapped).
23fn pci_read_config_io(bus: u8, dev: u8, func: u8, reg: u8) -> u32 {
24    let address = 0x8000_0000u32
25        | (bus as u32) << 16
26        | (dev as u32) << 11
27        | (func as u32) << 8
28        | (reg as u32 & 0xFC);
29    unsafe {
30        asm!(
31            "out dx, eax",
32            in("dx") 0xCF8u16,
33            in("eax") address,
34            options(nostack, preserves_flags)
35        );
36        let value: u32;
37        asm!(
38            "in eax, dx",
39            out("eax") value,
40            in("dx") 0xCFCu16,
41            options(nostack, preserves_flags)
42        );
43        value
44    }
45}
46
47/// The physical address of the PCIe ECAM, read from the MCH's `PCIEXBAR`
48/// register (which OVMF may have reprogrammed).
49fn ecam_base() -> usize {
50    // MCH is bus 0, device 0, function 0; PCIEXBAR is at config offset 0x60.
51    let low = pci_read_config_io(0, 0, 0, 0x60);
52    let high = pci_read_config_io(0, 0, 0, 0x64);
53    let value = ((high as u64) << 32) | low as u64;
54    info!("[pci] MCH PCIEXBAR = {:#x}", value);
55    assert_ne!(value & 1, 0, "PCIEXBAR not enabled by the firmware");
56    (value & 0xFFFF_FFFF_FFFF_FF00) as usize
57}
58
59pub struct Pci {
60    root: PciRoot,
61}
62
63impl Pci {
64    /// Map the ECAM window and wrap it in a `PciRoot`.
65    pub fn new() -> Self {
66        let ecam = ecam_base();
67        info!("[pci] ECAM at {:#x}", ecam);
68        ioremap(ecam, ECAM_BUS0_SIZE);
69        let mmio_base = crate::mm::phys_to_virt(ecam) as *mut u8;
70        Self {
71            root: unsafe { PciRoot::new(mmio_base, Cam::Ecam) },
72        }
73    }
74    /// Find the first VirtIO block device on bus 0 and map its BARs into the
75    /// physmap window (so that `PciTransport::new` can reach them via
76    /// `H::phys_to_virt`). Returns its bus/device/function.
77    pub fn find_block(&mut self) -> Option<DeviceFunction> {
78        let mut found = None;
79        for (bdf, info) in self.root.enumerate_bus(0) {
80            if virtio_device_type(&info) == Some(DeviceType::Block) {
81                info!(
82                    "[pci] found virtio block at {:02x}:{:02x}.{} (device {:#06x})",
83                    bdf.bus, bdf.device, bdf.function, info.device_id
84                );
85                found = Some(bdf);
86                break;
87            }
88        }
89        let bdf = found?;
90        let mut bar_index = 0u8;
91        while bar_index < 6 {
92            match self.root.bar_info(bdf, bar_index) {
93                Ok(BarInfo::Memory {
94                    address_type,
95                    address,
96                    size,
97                    ..
98                }) if size != 0 && address != 0 => {
99                    let start = (address as usize) & !(0xfff);
100                    let end = ((address as usize) + size as usize + 0xfff) & !(0xfff);
101                    ioremap(start, end - start);
102                    if address_type == MemoryBarType::Width64 {
103                        // 64-bit BAR occupies two consecutive slots; skip the
104                        // upper half so it is not treated as a separate BAR.
105                        bar_index += 2;
106                    } else {
107                        bar_index += 1;
108                    }
109                }
110                Ok(_) => bar_index += 1,
111                Err(_) => bar_index += 1,
112            }
113        }
114        Some(bdf)
115    }
116    pub fn root(&mut self) -> &mut PciRoot {
117        &mut self.root
118    }
119}
