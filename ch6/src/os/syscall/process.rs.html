1//! Process management syscalls
2use crate::fs::{OpenFlags, open_file};

9/// task exits and submit an exit code
10pub fn sys_exit(exit_code: i32) -> ! {
11    exit_current_and_run_next(exit_code)
12}
13
14/// current task gives up resources for other tasks
15pub fn sys_yield() -> isize {
16    suspend_current_and_run_next();
17    0
18}
19
20/// get time in milliseconds
21pub fn sys_get_time() -> isize {
22    get_time_ms() as isize
23}
24
25/// get the pid of the current task
26pub fn sys_getpid() -> isize {
27    current_task().unwrap().pid.0 as isize
28}
29
30/// create a child process, which is a copy of the current one
31pub fn sys_fork() -> isize {
32    let current_task = current_task().unwrap();
33    let new_task = current_task.fork();
34    let new_pid = new_task.pid.0;
35    // add new task to scheduler
36    add_task(new_task);
37    new_pid as isize
38}

40/// Read a NUL-terminated UTF-8 string from the user address space. The trap
41/// handler runs on the current task's own page table, so the user pointer is
42/// directly dereferenceable (the x86 analogue of `translated_str`).
43fn get_user_str(ptr: *const u8) -> &'static str {
44    let mut end = ptr;
45    unsafe {
46        while end.read_volatile() != b'\0' {
47            end = end.add(1);
48        }
49        let slice = core::slice::from_raw_parts(ptr, end as usize - ptr as usize);
50        core::str::from_utf8(slice).unwrap()
51    }
52}

54/// replace the current process with the program named by `path`, which is
55/// loaded from the file system
56pub fn sys_exec(path: *const u8) -> isize {
57    let path = get_user_str(path);
58    if let Some(app_inode) = open_file(path, OpenFlags::RDONLY) {
59        let all_data = app_inode.read_all();
60        let task = current_task().unwrap();
61        task.exec(all_data.as_slice());
62        0
63    } else {
64        -1
65    }
66}

68/// If there is not a child process whose pid is same as given, return -1.
69/// Else if there is a child process but it is still running, return -2.
70pub fn sys_waitpid(pid: isize, exit_code_ptr: *mut i32) -> isize {
71    let task = current_task().unwrap();
72    // find a child process
73
74    // ---- access current TCB exclusively
75    let mut inner = task.inner_exclusive_access();
76    if !inner
77        .children
78        .iter()
79        .any(|p| pid == -1 || pid as usize == p.getpid())
80    {
81        return -1;
82        // ---- release current PCB
83    }
84    let pair = inner.children.iter().enumerate().find(|(_, p)| {
85        // ++++ temporarily access child PCB lock exclusively
86        p.inner_exclusive_access().is_zombie() && (pid == -1 || pid as usize == p.getpid())
87        // ++++ release child PCB
88    });
89    if let Some((idx, _)) = pair {
90        let child = inner.children.remove(idx);
91        // confirm that child will be deallocated after removing from children list
92        assert_eq!(Arc::strong_count(&child), 1);
93        let found_pid = child.getpid();
94        // ++++ temporarily access child TCB exclusively
95        let exit_code = child.inner_exclusive_access().exit_code;
96        // ++++ release child PCB
97        // x86-64: the trap handler runs on this task's page table, so the user
98        // pointer is directly dereferenceable
99        unsafe {
100            *exit_code_ptr = exit_code;
101        }
102        found_pid as isize
103    } else {
104        -2
105    }
106    // ---- release current PCB lock automatically
107}
