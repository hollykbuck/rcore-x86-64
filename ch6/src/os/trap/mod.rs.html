1//! Trap handling functionality
2//!
3//! For rCore on x86-64, there are two entry points:
4//!
5//! - the `syscall` entry, whose address is stored in the `LSTAR` MSR. It
6//!   synthesizes an interrupt-like frame and forwards to [`trap_handler()`].
7//! - the exception stubs, pointed to by the IDT. The CPU switches to the
8//!   current task's kernel stack (the `rsp0` field of the TSS) and pushes the
9//!   interrupt frame, then each stub forwards to [`trap_handler()`].
10//!
11//! [`trap_handler()`] then calls different functionality based on what
12//! exactly the exception was. For example, syscalls go to [`syscall()`], and
13//! the APIC timer interrupt (vector 32) triggers task preemption.
14
15// Boot-time CPU structures are installed once in [`init()`] and then never
16// touched again, so `static mut` (via raw pointers) is fine here.
17#![allow(static_mut_refs)]
18// `trap.S` is written in GNU/AT&T syntax.
19#![allow(bad_asm_style)]

40static mut TSS: TaskStateSegment = TaskStateSegment::new();

42lazy_static! {
43    static ref GDT: Gdt = Gdt::new(core::ptr::addr_of!(TSS) as u64);
44    static ref IDT: InterruptDescriptorTable = {
45        let mut idt = InterruptDescriptorTable::new();
46        for i in 0..256 {
47            idt.set_handler(i, unsafe { exception_stub_addrs[i] } as u64);
48        }
49        idt
50    };
51}

53unsafe extern "C" {
54    /// the addresses of the 256 exception stubs, generated by `trap.S`
55    static exception_stub_addrs: [usize; 256];
56    safe fn syscall_entry();
57    safe fn trap_stack_top();
58    static mut current_stack_top: usize;
59}

61/// initialize the CPU (GDT, IDT, syscall MSRs). Page tables are taken over
62/// by [`crate::mm::init`].
63pub fn init() {
64    // set the kernel stack used when an interrupt switches from ring 3 to
65    // ring 0. The task manager overwrites this with the per-task kernel stack
66    // before the first task runs.
67    set_current_stack_top(linker_symbol_addr!(trap_stack_top) as u64);
68    GDT.load();
69    IDT.load();
70    msr::syscall_init(linker_symbol_addr!(syscall_entry) as u64);
71}

73/// Switch the kernel stack that the CPU uses for traps to `top`.
74///
75/// Both the `rsp0` field of the TSS (used for user-mode exceptions) and the
76/// `current_stack_top` global (used by `syscall_entry`) are updated. The task
77/// manager calls this right before switching to another task.
78pub fn set_current_stack_top(top: u64) {
79    unsafe {
80        (*core::ptr::addr_of_mut!(TSS)).set_rsp0(top);
81        current_stack_top = top as usize;
82    }
83}

85/// The timer interrupt is enabled by the user RFLAGS IF bit (see
86/// [`context::TrapContext::app_init_context`]) together with the APIC timer
87/// setup done by [`crate::timer::init`]. The kernel itself deliberately keeps
88/// IF cleared (interrupt gates and `SFMASK`) so that external interrupts only
89/// happen in user mode, mirroring how RISC-V ch3 only enables `stimer` in
90/// user mode. Nothing needs to be done here.
91pub fn enable_timer_interrupt() {}

93#[unsafe(no_mangle)]
94/// handle an interrupt, exception, or system call from user space
95pub fn trap_handler(cx: &mut TrapContext) -> &mut TrapContext {
96    match cx.trap_num {
97        TRAP_SYSCALL => {
98            let ret = syscall(cx.rax, [cx.rdi, cx.rsi, cx.rdx]);
99            // `cx` may have been replaced by `sys_exec` (it rebuilds the
100            // TrapContext on the kernel stack), so fetch it again
101            let cx = current_trap_cx();
102            cx.rax = ret as usize;
103        }
104        32 => {
105            // APIC timer interrupt
106            timer::timer_eoi();
107            timer::tick();
108            timer::set_next_trigger();
109            suspend_current_and_run_next();
110        }
111        14 => {
112            // page fault: CR2 holds the faulting linear address, error code
113            // bit 1 distinguishes a store (write) from a load (read) fault
114            let cr2: u64;
115            unsafe {
116                core::arch::asm!("mov rax, cr2", out("rax") cr2, options(nostack));
117            }
118            let access = if cx.error_code & 2 != 0 { "store" } else { "load" };
119            println!(
120                "[kernel] PageFault in application, {} addr = {:#x}, RIP = {:#x}, kernel killed it.",
121                access, cr2, cx.rip
122            );
123            exit_current_and_run_next(-2);
124        }
125        13 | 6 => {
126            println!("[kernel] IllegalInstruction in application, kernel killed it.");
127            exit_current_and_run_next(-3);
128        }
129        _ => {
130            panic!(
131                "Unsupported trap: vector = {}, error_code = {:#x}, RIP = {:#x}",
132                cx.trap_num, cx.error_code, cx.rip
133            );
134        }
135    }
136    cx
137}

139#[unsafe(no_mangle)]
140/// A trap that happened in kernel mode (i.e. a kernel bug): report and halt.
141pub extern "C" fn kernel_exception_panic(vector: usize, error_code: usize, fault_rip: usize) -> ! {
142    let cr2: u64;
143    unsafe {
144        core::arch::asm!("mov rax, cr2", out("rax") cr2, options(nostack));
145    }
146    error!(
147        "[kernel] Exception #{} (error_code = {:#x}) occurred in kernel mode, CR2 = {:#x}, RIP = {:#x}, kernel halted.",
148        vector, error_code, cr2, fault_rip
149    );
150    crate::uart::shutdown(true)
151}
