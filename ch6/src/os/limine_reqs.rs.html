1//! The limine boot protocol requests and the kernel entry point
2//!
3//! Limine boot a kernel as a raw binary (see `limine.conf`). It scans the
4//! kernel image for the request tags emitted in the `.limine_requests`
5//! section, fills in the corresponding responses, and finally jumps to
6//! `_start`. All of the requests and the two markers must be placed inside
7//! the `.limine_requests` section (which is `KEEP`ed by `linker.ld`).
8
9use core::arch::global_asm;
10
11global_asm!(
12    ".pushsection .limine_requests, \"a\"",
13    ".balign 8",
14
15    // LIMINE_REQUESTS_START_MARKER
16    ".quad 0xf6b8f4b39de7d1ae",
17    ".quad 0xfab91a6940fcb9cf",
18    ".quad 0x785c6ed015d3e316",
19    ".quad 0x181e920a7852b9d9",
20
21    // LIMINE_BASE_REVISION(6)
22    ".quad 0xf9562b2d5c95a6c8",
23    ".quad 0x6a7b384944536bdc",
24    ".quad 6",
25
26    // LIMINE_HHDM_REQUEST
27    ".global limine_hhdm_request",
28    "limine_hhdm_request:",
29    ".quad 0xc7b1dd30df4c8b88",
30    ".quad 0x0a82e883a194f07b",
31    ".quad 0x48dcf1cb8ad2b852",
32    ".quad 0x63984e959a98244b",
33    ".quad 0",
34    ".quad 0",
35
36    // LIMINE_MEMMAP_REQUEST
37    ".global limine_memmap_request",
38    "limine_memmap_request:",
39    ".quad 0xc7b1dd30df4c8b88",
40    ".quad 0x0a82e883a194f07b",
41    ".quad 0x67cf3d9d378a806f",
42    ".quad 0xe304acdfc50c3c62",
43    ".quad 0",
44    ".quad 0",
45
46    // LIMINE_EXECUTABLE_ADDRESS_REQUEST
47    ".global limine_executable_address_request",
48    "limine_executable_address_request:",
49    ".quad 0xc7b1dd30df4c8b88",
50    ".quad 0x0a82e883a194f07b",
51    ".quad 0x71ba76863cc55f63",
52    ".quad 0xb2644a48c516a487",
53    ".quad 0",
54    ".quad 0",
55
56    // LIMINE_REQUESTS_END_MARKER
57    ".quad 0xadc0e0531bb10d03",
58    ".quad 0x9572709f31764c62",
59
60    ".popsection",
61
62    // Stack symbols for logging (Limine creates a real stack for us)
63    ".section .bss.stack, \"aw\", @nobits",
64    ".balign 16",
65    ".globl boot_stack_lower_bound",
66    "boot_stack_lower_bound:",
67    ".space 32768",
68    ".globl boot_stack_top",
69    "boot_stack_top:",
70
71    ".section .text",
72    ".global _start",
73    "_start:",
74    "    lea rsp, [rip + boot_stack_top]",
75    "    call rust_main",
76);

78/// The response of `limine_hhdm_request`
79#[repr(C)]
80pub struct LimineHhdmResponse {
81    pub revision: u64,
82    /// offset of the higher-half direct map, `phys + offset == virt`
83    pub offset: u64,
84}
85
86/// The request for the higher-half direct map
87#[repr(C)]
88pub struct LimineHhdmRequest {
89    pub id: [u64; 4],
90    pub revision: u64,
91    pub response: Option<&'static mut LimineHhdmResponse>,
92}

94/// A memory map entry
95#[repr(C)]
96pub struct LimineMemmapEntry {

102/// The response of `limine_memmap_request`

134unsafe extern "C" {
135    static mut limine_hhdm_request: LimineHhdmRequest;
136    static mut limine_memmap_request: LimineMemmapRequest;
137    static mut limine_executable_address_request: LimineExecutableAddressRequest;
138}

140/// Get the higher-half direct map offset. Limine maps all physical memory at
141/// `phys + offset`, which is how we can touch arbitrary physical addresses
142/// before taking over paging.
143pub fn hhdm_offset() -> u64 {
144    let req = core::ptr::addr_of_mut!(limine_hhdm_request);
145    unsafe { (*req).response.as_ref() }
146        .expect("hhdm request not responded")
147        .offset
148}
149
150/// The physical address of the kernel image, reported by Limine. On Limine
151/// 12.5.2 this exactly matches the address found by walking Limine's own page
152/// tables (see HANDOVER §5.2), so the boot-time `translate()` walk is not
153/// needed anymore.
154pub fn kernel_physical_base() -> u64 {
155    let req = core::ptr::addr_of_mut!(limine_executable_address_request);
156    unsafe { (*req).response.as_ref() }
157        .expect("executable_address request not responded")
158        .physical_base
159}
160
161/// Iterate over the memory map entries reported by Limine.
162pub fn memmap_entries() -> &'static [&'static mut LimineMemmapEntry] {
163    let req = core::ptr::addr_of_mut!(limine_memmap_request);
164    let resp = unsafe { (*req).response.as_ref() }.expect("memmap request not responded");
165    let ptr = resp.entries;
166    assert!(!ptr.is_null(), "memmap response entries missing");
167    unsafe {
168        core::slice::from_raw_parts_mut(
169            ptr as *mut &'static mut LimineMemmapEntry,
170            resp.entry_count as usize,
171        )
172    }
