1//! Timer setup (local APIC timer for both preemption and timekeeping)
2//!
3//! Limine masks both the 8259 PICs and the IO-APIC before handing over, and
4//! the legacy PIT path is unreliable under QEMU's TCG emulation (interrupts
5//! are easily lost when the handler is slow). The local APIC timer, on the
6//! other hand, fires as reliably as the RISC-V CLINT timer, so it is used for
7//! both the time-slice rotation (round-robin preemption) and as the time
8//! source.
9//!
10//! The preemption rate is set to 1000 Hz. QEMU's TCG only advances its
11//! virtual clock in real time or under `-icount`, so `-icount shift=auto` is
12//! recommended to make the time slices evenly instruction-budgeted rather
13//! than host-load dependent (see HANDOVER §5.5).
14//!
15//! Timekeeping reconstructs a free-running monotonic counter from the APIC
16//! timer, the x86-64 analogue of the RISC-V `mtime`:
17//!
18//! - In periodic mode the current-count register (FEE0 0390H / x2APIC MSR
19//!   0x839) is automatically reloaded from the initial-count register each
20//!   time the count reaches zero (Intel SDM Vol. 3A §13.5.4).
21//! - So `elapsed_ticks = wraps * INITIAL_COUNT + (INITIAL_COUNT - current)`
22//!   is a free-running monotonic counter at `APIC_TIMER_FREQ` ticks/second.
23//! - The kernel runs with IF=0, so the timer interrupt cannot preempt the two
24//!   reads inside `elapsed_ticks()`: `wraps` and `current` are always
25//!   mutually consistent, and the counter is inherently monotonic.
26//!
27//! The APIC timer input frequency is exactly 1 GHz on QEMU (measured: a count
28//! of 10,000,000 gives 100 Hz). On real hardware it is the bus/crystal clock
29//! (SDM §13.5.4), so the hardcoded 1 GHz only holds for QEMU -- fine for this
30//! tutorial. No RTC is involved anymore: the counter is guest-side and
31//! inherently monotonic, so the previous RTC + minute-wrap bookkeeping is
32//! gone.
33
34use core::arch::asm;
35use core::sync::atomic::{AtomicUsize, Ordering};
36use log::*;
37
38/// number of timer interrupts per second (the preemption rate)
39const TICKS_PER_SEC: usize = 1000;
40/// the APIC timer input frequency, measured on QEMU (1 GHz)
41const APIC_TIMER_FREQ: usize = 1_000_000_000;
42/// the count programmed into the APIC timer (period = INITIAL_COUNT ns)
43const INITIAL_COUNT: usize = APIC_TIMER_FREQ / TICKS_PER_SEC;
44
45/// the MSR holding the local APIC base/enable state
46const APIC_BASE_MSR: u32 = 0x1B;
47/// x2APIC enable bit in the APIC base MSR
48const APIC_BASE_X2: u64 = 1 << 10;
49
50/// LAPIC register offsets
51const LAPIC_EOI: u32 = 0x0B0;
52const LAPIC_LVT_TIMER: u32 = 0x320;
53const LAPIC_TIMER_INIT_COUNT: u32 = 0x380;
54const LAPIC_TIMER_CURRENT_COUNT: u32 = 0x390;
55const LAPIC_TIMER_DIVIDE: u32 = 0x3E0;
56
57/// number of completed timer periods (interrupts delivered) since boot
58static WRAPS: AtomicUsize = AtomicUsize::new(0);
59
60unsafe fn rdmsr(msr: u32) -> u64 {
61    let mut hi: u32;
62    let mut lo: u32;
63    unsafe {
64        asm!(
65            "rdmsr",
66            in("ecx") msr,
67            out("eax") lo,
68            out("edx") hi,
69            options(nostack)
70        );
71    }
72    ((hi as u64) << 32) | lo as u64
73}
74
75unsafe fn wrmsr(msr: u32, value: u64) {
76    unsafe {
77        asm!(
78            "wrmsr",
79            in("ecx") msr,
80            in("eax") value as u32,
81            in("edx") (value >> 32) as u32,
82            options(nostack)
83        );
84    }
85}

87/// Write a local APIC register. Works in both xAPIC (MMIO) and x2APIC (MSR)
88/// modes.
89fn lapic_write(offset: u32, value: u32) {
90    unsafe {
91        let apic_base = rdmsr(APIC_BASE_MSR);
92        if apic_base & APIC_BASE_X2 != 0 {
93            wrmsr(0x800 + offset, value as u64);
94        } else {
95            // xAPIC MMIO: reach the physical window through the physmap
96            let base = crate::mm::phys_to_virt((apic_base & !0xFFF) as usize) as *mut u32;
97            core::ptr::write_volatile(base.add((offset / 4) as usize), value);
98        }
99    }
100}
101
102/// Read a local APIC register. Works in both xAPIC (MMIO) and x2APIC (MSR)
103/// modes.
104fn lapic_read(offset: u32) -> u32 {
105    unsafe {
106        let apic_base = rdmsr(APIC_BASE_MSR);
107        if apic_base & APIC_BASE_X2 != 0 {
108            rdmsr(0x800 + offset) as u32
109        } else {
110            let base = crate::mm::phys_to_virt((apic_base & !0xFFF) as usize) as *const u32;
111            core::ptr::read_volatile(base.add((offset / 4) as usize))
112        }
113    }
114}

116/// A free-running monotonic tick counter, the analogue of the RISC-V `mtime`.
117///
118/// Reconstructed from the periodic APIC timer: `wraps` counts full periods
119/// delivered by the interrupt handler, and `INITIAL_COUNT - current` is the
120/// number of ticks elapsed within the current period.
121fn elapsed_ticks() -> usize {
122    let current = lapic_read(LAPIC_TIMER_CURRENT_COUNT) as usize;
123    WRAPS.load(Ordering::Relaxed) * INITIAL_COUNT + (INITIAL_COUNT - current)
124}
125
126/// Initialize the timer: program the APIC timer in periodic mode to interrupt
127/// at `TICKS_PER_SEC` Hz on vector 32.
128pub fn init() {
129    // divide by 1
130    lapic_write(LAPIC_TIMER_DIVIDE, 0x0B);
131    // periodic mode, vector 32, unmasked
132    lapic_write(LAPIC_LVT_TIMER, 0x20 | (1 << 17));
133    // count for TICKS_PER_SEC Hz
134    lapic_write(LAPIC_TIMER_INIT_COUNT, INITIAL_COUNT as u32);
135    info!("timer: APIC timer interrupt at ~{} Hz", TICKS_PER_SEC);
136}
137
138/// Count one timer interrupt (one full period elapsed). Called from the timer
139/// interrupt handler.
140pub fn tick() {
141    WRAPS.fetch_add(1, Ordering::Relaxed);
142}
143
144/// Send the End-Of-Interrupt to the local APIC. Must be called at the start
145/// of the timer handler, otherwise no further interrupt is delivered.
146pub fn timer_eoi() {
147    lapic_write(LAPIC_EOI, 0);
148}
149
150/// get current time in milliseconds
151///
152/// Derived from the monotonic APIC-timer counter, so the resolution is 1 ms
153/// (and monotonic by construction).
154pub fn get_time_ms() -> usize {
155    elapsed_ticks() / (APIC_TIMER_FREQ / 1000)
156}
