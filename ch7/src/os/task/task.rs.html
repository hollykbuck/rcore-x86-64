22pub struct TaskControlBlockInner {
23    /// virtual address of the `TrapContext` on this process's kernel stack.
24    /// The kernel stack lives in the shared kernel high-half mapping, so this
25    /// address is valid under any active page table.
26    pub trap_cx_ptr: usize,
27    #[allow(unused)]
28    pub base_size: usize,
29    pub task_cx: TaskContext,
30    pub task_status: TaskStatus,
31    pub memory_set: MemorySet,
32    pub parent: Option<Weak<TaskControlBlock>>,
33    pub children: Vec<Arc<TaskControlBlock>>,
34    pub exit_code: i32,
35    pub fd_table: Vec<Option<Arc<dyn File + Send + Sync>>>,
36    /// signals received by the current process (pending)
37    pub signals: SignalFlags,
38    /// signals that should not be responded to
39    pub signal_mask: SignalFlags,
40    /// the signal currently being handled by the user handler, -1 if none
41    pub handling_sig: isize,
42    /// the actions of the current process for all signals
43    pub signal_actions: SignalActions,
44    /// true if the task is killed by a signal (or SIGKILL)
45    pub killed: bool,
46    /// true if the task is frozen by SIGSTOP, waiting for SIGCONT
47    pub frozen: bool,
48    /// the trap context saved when a user signal handler is invoked, restored
49    /// by `sigreturn`
50    pub trap_ctx_backup: Option<TrapContext>,
51}

53impl TaskControlBlockInner {
54    pub fn get_trap_cx(&self) -> &'static mut TrapContext {
55        unsafe { &mut *(self.trap_cx_ptr as *mut TrapContext) }
56    }
57    pub fn get_user_token(&self) -> usize {
58        self.memory_set.token()
59    }
60    fn get_status(&self) -> TaskStatus {
61        self.task_status
62    }
63    pub fn is_zombie(&self) -> bool {
64        self.get_status() == TaskStatus::Zombie
65    }
66    pub fn alloc_fd(&mut self) -> usize {
67        if let Some(fd) = (0..self.fd_table.len()).find(|fd| self.fd_table[*fd].is_none()) {
68            fd
69        } else {
70            self.fd_table.push(None);
71            self.fd_table.len() - 1
72        }
73    }
74}

76impl TaskControlBlock {
77    pub fn inner_exclusive_access(&self) -> RefMut<'_, TaskControlBlockInner> {
78        self.inner.exclusive_access()
79    }
80    pub fn new(elf_data: &[u8]) -> Self {
81        // memory_set with elf program headers/user stack
82        let (memory_set, user_sp, entry_point) = MemorySet::from_elf(elf_data);
83        // alloc a pid and a kernel stack in the shared kernel high-half
84        let pid_handle = pid_alloc();
85        let kernel_stack = KernelStack::new(&pid_handle);
86        let trap_cx_ptr = kernel_stack.push_context(TrapContext::app_init_context(
87            entry_point,
88            user_sp,
89        ));
90        let task_control_block = Self {
91            pid: pid_handle,
92            kernel_stack,
93            inner: unsafe {
94                UPSafeCell::new(TaskControlBlockInner {
95                    trap_cx_ptr,
96                    base_size: user_sp,
97                    task_cx: TaskContext::goto_restore(trap_cx_ptr),
98                    task_status: TaskStatus::Ready,
99                    memory_set,
100                    parent: None,
101                    children: Vec::new(),
102                    exit_code: 0,
103                    // 0 -> stdin, 1 -> stdout, 2 -> stderr
104                    fd_table: vec![
105                        Some(Arc::new(Stdin)),
106                        Some(Arc::new(Stdout)),
107                        Some(Arc::new(Stdout)),
108                    ],
109                    signals: SignalFlags::empty(),
110                    signal_mask: SignalFlags::empty(),
111                    handling_sig: -1,
112                    signal_actions: SignalActions::default(),
113                    killed: false,
114                    frozen: false,
115                    trap_ctx_backup: None,
116                })
117            },
118        };
119        task_control_block
120    }
121    /// Load a new program image into this process (the `exec` syscall).
122    ///
123    /// x86-64 note: the trap entry does not switch CR3, so the new address
124    /// space must be activated here; otherwise the `iretq` back to user mode
125    /// would execute the new entry point under the old page table.
126    pub fn exec(&self, elf_data: &[u8], args: Vec<String>) {
127        // memory_set with elf program headers/user stack
128        let (memory_set, mut user_sp, entry_point) = MemorySet::from_elf(elf_data);
129        // push the command line arguments on the user stack, in the same
130        // layout as the RISC-V tutorial: a NUL-terminated argv pointer array
131        // followed by each argument string. The entry gets `argc`/`argv` in
132        // RDI/RSI (x86-64 SysV: first/second argument registers).
133        user_sp -= (args.len() + 1) * core::mem::size_of::<usize>();
134        let argv_base = user_sp;
135        let mut argv: Vec<_> = (0..=args.len())
136            .map(|arg| {
137                crate::mm::translated_refmut(
138                    memory_set.token(),
139                    (argv_base + arg * core::mem::size_of::<usize>()) as *mut usize,
140                )
141            })
142            .collect();
143        *argv[args.len()] = 0;
144        for i in 0..args.len() {
145            user_sp -= args[i].len() + 1;
146            *argv[i] = user_sp;
147            let mut p = user_sp;
148            for c in args[i].as_bytes() {
149                *crate::mm::translated_refmut(memory_set.token(), p as *mut u8) = *c;
150                p += 1;
151            }
152            *crate::mm::translated_refmut(memory_set.token(), p as *mut u8) = 0;
153        }
154        // keep the user stack 16-byte aligned (x86-64 SysV ABI)
155        user_sp -= user_sp % 16;
156        let mut inner = self.inner_exclusive_access();
157        // substitute the memory set (the old one is dropped, freeing frames)
158        inner.memory_set = memory_set;
159        // initialize base_size
160        inner.base_size = user_sp;
161        // rebuild the TrapContext on the same kernel stack
162        let mut trap_cx = TrapContext::app_init_context(entry_point, user_sp);
163        trap_cx.rdi = args.len();
164        trap_cx.rsi = argv_base;
165        let trap_cx_ptr = self.kernel_stack.push_context(trap_cx);
166        inner.trap_cx_ptr = trap_cx_ptr;
167        // switch to the new address space now
168        inner.memory_set.activate();
169    }
170    /// Create a new process by copying the current one (the `fork` syscall).
171    ///
172    /// x86-64 note: the TrapContext lives on the kernel stack (not in user
173    /// memory like the RISC-V `TRAP_CONTEXT` page), so the child gets an
174    /// explicit copy of the parent's current trap context on its own kernel
175    /// stack, with `rax` zeroed so `fork` returns 0 in the child.
176    pub fn fork(self: &Arc<Self>) -> Arc<Self> {
177        // ---- access parent PCB exclusively
178        let mut parent_inner = self.inner_exclusive_access();
179        // copy user space
180        let memory_set = MemorySet::from_existed_user(&parent_inner.memory_set);
181        // alloc a pid and a kernel stack in kernel space
182        let pid_handle = pid_alloc();
183        let kernel_stack = KernelStack::new(&pid_handle);
184        // copy the parent's current TrapContext, with rax = 0 for the child
185        let mut child_trap_cx = *parent_inner.get_trap_cx();
186        child_trap_cx.rax = 0;
187        let trap_cx_ptr = kernel_stack.push_context(child_trap_cx);
188        // copy the parent's fd table (sharing the underlying files)
189        let mut new_fd_table: Vec<Option<Arc<dyn File + Send + Sync>>> = Vec::new();
190        for fd in parent_inner.fd_table.iter() {
191            new_fd_table.push(fd.clone());
192        }
193        let task_control_block = Arc::new(TaskControlBlock {
194            pid: pid_handle,
195            kernel_stack,
196            inner: unsafe {
197                UPSafeCell::new(TaskControlBlockInner {
198                    trap_cx_ptr,
199                    base_size: parent_inner.base_size,
200                    task_cx: TaskContext::goto_restore(trap_cx_ptr),
201                    task_status: TaskStatus::Ready,
202                    memory_set,
203                    parent: Some(Arc::downgrade(self)),
204                    children: Vec::new(),
205                    exit_code: 0,
206                    fd_table: new_fd_table,
207                    signals: SignalFlags::empty(),
208                    // inherit the signal_mask and signal_action
209                    signal_mask: parent_inner.signal_mask,
210                    handling_sig: -1,
211                    signal_actions: parent_inner.signal_actions.clone(),
212                    killed: false,
213                    frozen: false,
214                    trap_ctx_backup: None,
215                })
216            },
217        });
218        // add child
219        parent_inner.children.push(task_control_block.clone());
220        // return
221        task_control_block
222        // ---- release parent PCB automatically
223    }
224    pub fn getpid(&self) -> usize {
225        self.pid.0
226    }
227}
