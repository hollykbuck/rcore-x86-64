1//! Implementation of [`Processor`] and the intersection of control flow.
2//!
3//! The scheduler is an idle control flow (`run_tasks` loop) that fetches
4//! ready tasks and `__switch`es to them; `schedule` switches back to the idle
5//! control flow when the current task suspends or exits. On x86-64, every
6//! switch to a task must (1) activate its page table (`mov cr3`) and (2) point
7//! the trap kernel stack (`TSS.rsp0` + `current_stack_top`) at its kernel
8//! stack, because the CPU does not switch address spaces on trap.
9
10use super::__switch;
11use super::{TaskContext, TaskControlBlock};
12use super::{TaskStatus, fetch_task};
13use crate::sync::UPSafeCell;
14use crate::trap::{TrapContext, set_current_stack_top};
15use alloc::sync::Arc;
16use lazy_static::*;
17
18/// Processor management structure
19pub struct Processor {
20    /// The task currently executing on the current processor
21    current: Option<Arc<TaskControlBlock>>,
22    /// The basic control flow of each core, helping to select and switch process
23    idle_task_cx: TaskContext,
24}
25
26impl Processor {
27    /// Create an empty Processor
28    pub fn new() -> Self {
29        Self {
30            current: None,
31            idle_task_cx: TaskContext::zero_init(),
32        }
33    }
34    /// Get mutable reference to `idle_task_cx`
35    fn get_idle_task_cx_ptr(&mut self) -> *mut TaskContext {
36        &mut self.idle_task_cx as *mut _
37    }
38    /// Get current task in moving semantics
39    pub fn take_current(&mut self) -> Option<Arc<TaskControlBlock>> {
40        self.current.take()
41    }
42    /// Get current task in cloning semantics
43    pub fn current(&self) -> Option<Arc<TaskControlBlock>> {
44        self.current.as_ref().map(Arc::clone)
45    }
46}
47
48lazy_static! {
49    pub static ref PROCESSOR: UPSafeCell<Processor> =
50        unsafe { UPSafeCell::new(Processor::new()) };
51}
52
53/// The main part of process execution and scheduling.
54///
55/// Loop `fetch_task` to get the process that needs to run, and switch to it
56/// through `__switch`. The idle control flow runs on the boot stack.
57pub fn run_tasks() -> ! {
58    loop {
59        let mut processor = PROCESSOR.exclusive_access();
60        if let Some(task) = fetch_task() {
61            let idle_task_cx_ptr = processor.get_idle_task_cx_ptr();
62            // access the coming task TCB exclusively
63            let mut task_inner = task.inner_exclusive_access();
64            let next_task_cx_ptr = &task_inner.task_cx as *const TaskContext;
65            task_inner.task_status = TaskStatus::Running;
66            // x86-64: activate the task's page table before switching to it
67            // (this is the only place CR3 changes; traps never switch it).
68            task_inner.memory_set.activate();
69            drop(task_inner);
70            // x86-64: point the trap kernel stack at the task's own stack so
71            // its first user-mode trap lands on the right stack.
72            set_current_stack_top(task.kernel_stack.get_top() as u64);
73            // release the coming task TCB manually
74            processor.current = Some(task);
75            // release the processor manually
76            drop(processor);
77            unsafe {
78                __switch(idle_task_cx_ptr, next_task_cx_ptr);
79            }
80        }
81    }
82}

84/// Take the current task, leaving a `None` in its place
85pub fn take_current_task() -> Option<Arc<TaskControlBlock>> {
86    PROCESSOR.exclusive_access().take_current()
87}
88
89/// Get the running task
90pub fn current_task() -> Option<Arc<TaskControlBlock>> {
91    PROCESSOR.exclusive_access().current()
92}
93
94/// Get the token of the address space of the current task
95pub fn current_user_token() -> usize {
96    let task = current_task().unwrap();
97    let token = task.inner_exclusive_access().get_user_token();
98    token
99}
100
101/// Get the mutable reference to the trap context of the current task
102pub fn current_trap_cx() -> &'static mut TrapContext {
103    current_task()
104        .unwrap()
105        .inner_exclusive_access()
106        .get_trap_cx()
107}
108
109/// Return to the idle control flow for new scheduling
110pub fn schedule(switched_task_cx_ptr: *mut TaskContext) {
111    let mut processor = PROCESSOR.exclusive_access();
112    let idle_task_cx_ptr = processor.get_idle_task_cx_ptr();
113    drop(processor);
114    unsafe {
115        __switch(switched_task_cx_ptr, idle_task_cx_ptr);
116    }
