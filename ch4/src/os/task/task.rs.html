1//! Types related to task management
2use super::TaskContext;
3use crate::mm::{MemorySet, VirtAddr};
4use crate::trap::TrapContext;
5
6/// task control block structure
7pub struct TaskControlBlock {
8    pub task_status: TaskStatus,
9    pub task_cx: TaskContext,
10    pub memory_set: MemorySet,
11    /// virtual address of the `TrapContext` pushed on this task's kernel
12    /// stack. The kernel stack lives in the shared kernel high-half mapping,
13    /// so this address is valid under any active page table.
14    pub trap_cx_ptr: usize,
15    #[allow(unused)]
16    pub base_size: usize,
17    pub heap_bottom: usize,
18    pub program_brk: usize,
19}
20
21impl TaskControlBlock {
22    pub fn get_trap_cx(&self) -> &'static mut TrapContext {
23        unsafe { &mut *(self.trap_cx_ptr as *mut TrapContext) }
24    }
25    pub fn get_user_token(&self) -> usize {
26        self.memory_set.token()
27    }
28    pub fn new(elf_data: &[u8], app_id: usize) -> Self {
29        // memory_set with elf program headers/user stack
30        let (memory_set, user_sp, entry_point) = MemorySet::from_elf(elf_data);
31        let task_status = TaskStatus::Ready;
32        // prepare a TrapContext on the task's kernel stack
33        let trap_cx_ptr = crate::loader::push_context(
34            app_id,
35            TrapContext::app_init_context(entry_point, user_sp),
36        );
37        let task_control_block = Self {
38            task_status,
39            task_cx: TaskContext::goto_restore(trap_cx_ptr),
40            memory_set,
41            trap_cx_ptr,
42            base_size: user_sp,
43            heap_bottom: user_sp,
44            program_brk: user_sp,
45        };
46        task_control_block
47    }
48    /// change the location of the program break. return None if failed.
49    pub fn change_program_brk(&mut self, size: i32) -> Option<usize> {
50        let old_break = self.program_brk;
51        let new_brk = self.program_brk as isize + size as isize;
52        if new_brk < self.heap_bottom as isize {
53            return None;
54        }
55        let result = if size < 0 {
56            self.memory_set
57                .shrink_to(VirtAddr(self.heap_bottom), VirtAddr(new_brk as usize))
58        } else {
59            self.memory_set
60                .append_to(VirtAddr(self.heap_bottom), VirtAddr(new_brk as usize))
61        };
62        if result {
63            self.program_brk = new_brk as usize;
64            Some(old_break)
65        } else {
66            None
67        }
68    }
69}

72/// task status: Ready, Running, Exited
73pub enum TaskStatus {
74    Ready,
75    Running,
76    Exited,
