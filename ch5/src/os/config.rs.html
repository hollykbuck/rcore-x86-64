3pub const USER_STACK_SIZE: usize = 4096 * 2;
4pub const KERNEL_STACK_SIZE: usize = 4096 * 2;
5pub const KERNEL_HEAP_SIZE: usize = 0x30_0000;
6pub const PAGE_SIZE: usize = 0x1000;
7pub const PAGE_SIZE_BITS: usize = 0xc;
8
9/// The virtual address where the kernel image is linked
10pub const KERNEL_BASE: usize = 0xffffffff80000000;
11/// Fallback physical memory size (QEMU `-m 256M`), used if the Limine memory
12/// map reports no usable region.
13pub const MEMORY_END: usize = 0x1000_0000;
14/// The physical base of the local APIC MMIO window
15pub const LAPIC_BASE: usize = 0xFEE0_0000;
16
17/// Start of the per-process kernel stack region. This is the *start* of
18/// `PML4[511]`/`PDPT[511]`, one PDPT entry above the kernel image (`PDPT[510]`).
19pub const KERNEL_STACK_BASE: usize = 0xffffffffc0000000;
20
21/// Return (bottom, top) of the kernel stack of the process with `pid`.
22///
23/// The layout **ascends** from `KERNEL_STACK_BASE` (one guard page between
24/// stacks) so it stays entirely within `PDPT[511]`. Descending from the base
25/// would fall into `PDPT[510]` (where the kernel image lives) -- verified with
26/// `pid = 1` landing at `0xffffffffbfffd000`.
27pub fn kernel_stack_position(pid: usize) -> (usize, usize) {
28    let bottom = KERNEL_STACK_BASE + pid * (KERNEL_STACK_SIZE + PAGE_SIZE);
29    (bottom, bottom + KERNEL_STACK_SIZE)
30}
