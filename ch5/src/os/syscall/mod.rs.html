1//! Implementation of syscalls
2//!
3//! The single entry point to all system calls, [`syscall()`], is called
4//! whenever userspace wishes to perform a system call using the `syscall`
5//! instruction. In this case, the processor jumps to the entry pointed to by
6//! the `LSTAR` MSR, which is handled as one of the cases in
7//! [`crate::trap::trap_handler`].
8//!
9//! For clarity, each single syscall is implemented as its own function, named
10//! `sys_` then the name of the syscall. You can find functions like this in
11//! submodules, and you should also implement syscalls this way.
12
13// Syscall numbers follow the Linux x86-64 convention where one exists.
14const SYSCALL_READ: usize = 0;
15const SYSCALL_WRITE: usize = 1;
16const SYSCALL_GETPID: usize = 39;
17const SYSCALL_FORK: usize = 57;
18const SYSCALL_EXEC: usize = 59;
19const SYSCALL_EXIT: usize = 60;
20const SYSCALL_WAITPID: usize = 61;
21const SYSCALL_YIELD: usize = 24;
22/// `get_time` has no direct Linux x86-64 counterpart, so a teaching number is
23/// used here (201 happens to be Linux's `time`; we return milliseconds).
24const SYSCALL_GET_TIME: usize = 201;
25

32/// handle syscall exception with `syscall_id` and other arguments
33pub fn syscall(syscall_id: usize, args: [usize; 3]) -> isize {
34    match syscall_id {
35        SYSCALL_READ => sys_read(args[0], args[1] as *const u8, args[2]),
36        SYSCALL_WRITE => sys_write(args[0], args[1] as *const u8, args[2]),
37        SYSCALL_GETPID => sys_getpid(),
38        SYSCALL_FORK => sys_fork(),
39        SYSCALL_EXEC => sys_exec(args[0] as *const u8),
40        SYSCALL_EXIT => sys_exit(args[0] as i32),
41        SYSCALL_WAITPID => sys_waitpid(args[0] as isize, args[1] as *mut i32),
42        SYSCALL_YIELD => sys_yield(),
43        SYSCALL_GET_TIME => sys_get_time(),
44        _ => panic!("Unsupported syscall_id: {}", syscall_id),
45    }
46}
