1//! The Interrupt Descriptor Table
2//!
3//! Every one of the 256 vectors points to the corresponding stub in `trap.S`.
4//! Only the vectors that the kernel uses in practice are expected; any other
5//! exception is reported by [`crate::trap::trap_handler`] as unsupported.
6
7use super::gdt::KERNEL_CS;
8use core::arch::asm;
9
10const INTERRUPT_GATE: u8 = 0x8E; // present, DPL 0, interrupt gate
11
12/// A single IDT entry. An x86-64 IDT entry is exactly 16 bytes.
13#[repr(C, packed)]
14#[derive(Clone, Copy)]
15struct IdtEntry {
16    offset_low: u16,
17    selector: u16,
18    ist: u8,
19    type_attr: u8,
20    offset_mid: u16,
21    offset_high: u32,
22    reserved: u32,
23}
24
25impl IdtEntry {
26    const fn new(handler: u64, selector: u16, type_attr: u8) -> Self {
27        Self {
28            offset_low: handler as u16,
29            selector,
30            ist: 0,
31            type_attr,
32            offset_mid: (handler >> 16) as u16,
33            offset_high: (handler >> 32) as u32,
34            reserved: 0,
35        }
36    }
37}
38
39const fn empty() -> IdtEntry {
40    IdtEntry::new(0, 0, 0)
41}
42
43/// The IDT, 256 entries of 16 bytes
44#[repr(C, align(16))]
45pub struct InterruptDescriptorTable {
46    entries: [IdtEntry; 256],
47}
48
49impl InterruptDescriptorTable {
50    pub fn new() -> Self {
51        Self {
52            entries: [empty(); 256],
53        }
54    }
55
56    /// Set the handler address of the `vector`-th entry.
57    pub fn set_handler(&mut self, vector: usize, handler: u64) {
58        self.entries[vector] = IdtEntry::new(handler, KERNEL_CS, INTERRUPT_GATE);
59    }
60
61    /// Load the IDT with `lidt`.
62    pub fn load(&'static self) {
63        #[repr(C, packed)]
64        struct IdtPointer {
65            limit: u16,
66            base: u64,
67        }
68        let ptr = IdtPointer {
69            limit: (core::mem::size_of::<Self>() - 1) as u16,
70            base: self as *const _ as u64,
71        };
72        unsafe {
73            asm!("lidt [{ptr}]", ptr = in(reg) &ptr, options(readonly, nostack));
74        }
75    }
76}
