1//! Loading user applications into memory
2//!
3//! For chapter 3, user applications are simply part of the data included in the
4//! kernel binary, so we only need to copy them to the space allocated for each
5//! app to load them. We also allocate a fixed kernel stack for each task and
6//! compute a fixed user stack region for it.
7//!
8//! On x86-64 the kernel stacks live in the kernel `.bss` (supervisor-only,
9//! used by trap handling). The user stacks must be placed in the low
10//! identity-mapped, user-accessible memory, since the ring-3 applications
11//! cannot touch the kernel's pages.
12
13use crate::config::*;
14use crate::trap::TrapContext;
15
16#[repr(align(4096))]
17#[derive(Copy, Clone)]
18struct KernelStack {
19    data: [u8; KERNEL_STACK_SIZE],
20}
21
22static KERNEL_STACK: [KernelStack; MAX_APP_NUM] = [KernelStack {
23    data: [0; KERNEL_STACK_SIZE],
24}; MAX_APP_NUM];
25
26impl KernelStack {
27    fn get_sp(&self) -> usize {
28        self.data.as_ptr() as usize + KERNEL_STACK_SIZE
29    }
30    pub fn push_context(&self, trap_cx: TrapContext) -> usize {
31        let trap_cx_ptr = (self.get_sp() - core::mem::size_of::<TrapContext>()) as *mut TrapContext;
32        unsafe {
33            *trap_cx_ptr = trap_cx;
34        }
35        trap_cx_ptr as usize
36    }
37}
38
39/// Get the top of task `app_id`'s kernel stack. Must be kept in sync with the
40/// TSS `rsp0` (see [`crate::trap::set_current_stack_top`]).
41pub fn kernel_stack_top(app_id: usize) -> usize {
42    KERNEL_STACK[app_id].get_sp()
43}
44
45/// The user stack of task `app_id` lives right above the whole app area, in
46/// the low identity-mapped (user-accessible) memory.
47fn user_stack_top(app_id: usize) -> usize {
48    APP_BASE_ADDRESS + MAX_APP_NUM * APP_SIZE_LIMIT + (app_id + 1) * USER_STACK_SIZE
49}
50
51/// Get base address of app i.
52fn get_base_i(app_id: usize) -> usize {
53    APP_BASE_ADDRESS + app_id * APP_SIZE_LIMIT
54}
55
56/// Get the total number of applications.
57pub fn get_num_app() -> usize {
58    unsafe extern "C" {
59        safe fn _num_app();
60    }
61    unsafe { (linker_symbol_addr!(_num_app) as *const usize).read_volatile() }
62}
63
64/// Load nth user app at
65/// [APP_BASE_ADDRESS + n * APP_SIZE_LIMIT, APP_BASE_ADDRESS + (n+1) * APP_SIZE_LIMIT).
66pub fn load_apps() {
67    unsafe extern "C" {
68        safe fn _num_app();
69    }
70    let num_app_ptr = linker_symbol_addr!(_num_app) as *const usize;
71    let num_app = get_num_app();
72    let app_start = unsafe { core::slice::from_raw_parts(num_app_ptr.add(1), num_app + 1) };
73    // load apps
74    for i in 0..num_app {
75        let base_i = get_base_i(i);
76        // clear region
77        (base_i..base_i + APP_SIZE_LIMIT)
78            .for_each(|addr| unsafe { (addr as *mut u8).write_volatile(0) });
79        // load app from data section to memory
80        let src = unsafe {
81            core::slice::from_raw_parts(app_start[i] as *const u8, app_start[i + 1] - app_start[i])
82        };
83        let dst = unsafe { core::slice::from_raw_parts_mut(base_i as *mut u8, src.len()) };
84        dst.copy_from_slice(src);
85    }
86    // No instruction-fetch fence is needed on x86-64: the instruction cache is
87    // coherent with the data written here, and no instruction is fetched from
88    // these pages before the first `iretq` to the app anyway.
89}

91/// get app info with entry and sp and save `TrapContext` in kernel stack
92pub fn init_app_cx(app_id: usize) -> usize {
93    KERNEL_STACK[app_id].push_context(TrapContext::app_init_context(
94        get_base_i(app_id),
95        user_stack_top(app_id),
96    ))
97}
