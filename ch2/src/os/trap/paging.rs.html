1//! The page tables
2//!
3//! Limine hands the kernel over with its own page tables, which map the kernel
4//! (and all physical memory through the higher-half direct map) as
5//! supervisor-only pages. That is not enough for the user apps of this
6//! chapter, which must run in ring 3 without a dedicated address space.
7//!
8//! Therefore we install our own tiny page tables that:
9//!
10//! - identity map the low 4GiB with 2MiB pages, marked **user** accessible, so
11//!   that both the kernel and the ring-3 apps can reach the app memory;
12//! - map the kernel image at `0xffffffff80000000` with 4KiB pages, marked
13//!   supervisor-only, so that apps cannot touch kernel memory.
14//!
15//! The kernel is loaded by Limine at an arbitrary physical address, so we
16//! recover the exact physical location by walking Limine's own page tables
17//! (accessible through the higher-half direct map). This is equivalent to the
18//! `limine_executable_address_request` response on Limine 12.5.2 (both match,
19//! delta = 0) but does not depend on the ELF segment alignment logic.
20
21// The page tables are installed once at boot, so `static mut` (via raw
22// pointers where needed) is fine here.
23#![allow(static_mut_refs)]
24
25use core::arch::asm;
26
27const PAGE_PRESENT: u64 = 1 << 0;
28const PAGE_WRITABLE: u64 = 1 << 1;
29const PAGE_USER: u64 = 1 << 2;
30const PAGE_HUGE: u64 = 1 << 7; // PS bit for 2MiB/1GiB pages
31const PAGE_ADDR_MASK: u64 = 0x000F_FFFF_FFFF_F000; // physical address bits
32
33const SZ_4K: u64 = 0x1000;
34const SZ_2M: u64 = 0x200000;
35const SZ_1G: u64 = 0x40000000;
36
37const KERNEL_BASE: u64 = 0xffffffff80000000;
38
39/// A 4KiB-aligned page table (512 entries)
40#[repr(C, align(4096))]
41struct PageTable([u64; 512]);
42
43static mut PML4: PageTable = PageTable([0; 512]);
44static mut PDPT_LOW: PageTable = PageTable([0; 512]);
45static mut PDPT_KERNEL: PageTable = PageTable([0; 512]);
46static mut PD_LOW0: PageTable = PageTable([0; 512]);
47static mut PD_LOW1: PageTable = PageTable([0; 512]);
48static mut PD_LOW2: PageTable = PageTable([0; 512]);
49static mut PD_LOW3: PageTable = PageTable([0; 512]);
50static mut PD_KERNEL: PageTable = PageTable([0; 512]);
51static mut PT_KERNEL0: PageTable = PageTable([0; 512]);
52static mut PT_KERNEL1: PageTable = PageTable([0; 512]);
53static mut PT_KERNEL2: PageTable = PageTable([0; 512]);
54static mut PT_KERNEL3: PageTable = PageTable([0; 512]);
55
56unsafe fn read_cr3() -> u64 {
57    let cr3: u64;
58    unsafe { asm!("mov rax, cr3", out("rax") cr3, options(nostack)) };
59    cr3
60}
61
62/// Translate a virtual address to a physical one, by walking the currently
63/// active (i.e. Limine's) page tables. All physical accesses go through the
64/// higher-half direct map.
65fn translate(virt: u64) -> Option<u64> {
66    let hhdm = crate::limine_reqs::hhdm_offset();
67    // read the (physical) page table entry at `phys + 8 * index`
68    let read = |phys: u64, index: usize| -> u64 {
69        unsafe {
70            *((hhdm + phys + index as u64 * 8) as *const u64)
71        }
72    };
73
74    let pml4 = unsafe { read_cr3() };
75    let e = read(pml4, ((virt >> 39) & 0x1FF) as usize);
76    if e & PAGE_PRESENT == 0 {
77        return None;
78    }
79    let pdpt = e & PAGE_ADDR_MASK;
80    if e & PAGE_HUGE != 0 {
81        return Some((pdpt & !(SZ_1G - 1)) + (virt & (SZ_1G - 1)));
82    }
83
84    let e = read(pdpt, ((virt >> 30) & 0x1FF) as usize);
85    if e & PAGE_PRESENT == 0 {
86        return None;
87    }
88    let pd = e & PAGE_ADDR_MASK;
89    if e & PAGE_HUGE != 0 {
90        return Some((pd & !(SZ_2M - 1)) + (virt & (SZ_2M - 1)));
91    }
92
93    let e = read(pd, ((virt >> 21) & 0x1FF) as usize);
94    if e & PAGE_PRESENT == 0 {
95        return None;
96    }
97    let pt = e & PAGE_ADDR_MASK;
98    if e & PAGE_HUGE != 0 {
99        return Some((pt & !(SZ_2M - 1)) + (virt & (SZ_2M - 1)));
100    }
101
102    let e = read(pt, ((virt >> 12) & 0x1FF) as usize);
103    if e & PAGE_PRESENT == 0 {
104        return None;
105    }
106    Some((e & PAGE_ADDR_MASK) + (virt & (SZ_4K - 1)))
107}

109/// Set up the page tables and switch to them.
110pub fn init() {
111    unsafe {
112        // the physical address of a kernel symbol, given its virtual address
113        let phys_of = |virt: u64| translate(virt).expect("kernel address not mapped");
114
115        let kernel_phys = translate(KERNEL_BASE).expect("kernel not mapped by Limine");
116
117        // identity map the low 4GiB, user accessible. The first 2MiB page is
118        // left unmapped so that a null-pointer access from a user app faults
119        // (just like on the RISC-V port where address 0 is not backed by RAM).
120        for (i, pd) in [&mut PD_LOW0, &mut PD_LOW1, &mut PD_LOW2, &mut PD_LOW3]
121            .iter_mut()
122            .enumerate()
123        {
124            for (j, entry) in pd.0.iter_mut().enumerate() {
125                let page = i as u64 * 512 + j as u64;
126                if page == 0 {
127                    continue; // leave the 0..2MiB region unmapped
128                }
129                *entry = (i as u64 * SZ_1G + j as u64 * SZ_2M)
130                    | PAGE_PRESENT
131                    | PAGE_WRITABLE
132                    | PAGE_USER
133                    | PAGE_HUGE;
134            }
135        }
136        let pd_low = [
137            &PD_LOW0 as *const _ as u64,
138            &PD_LOW1 as *const _ as u64,
139            &PD_LOW2 as *const _ as u64,
140            &PD_LOW3 as *const _ as u64,
141        ];
142        for (i, pd) in pd_low.iter().enumerate() {
143            PDPT_LOW.0[i] = phys_of(*pd) | PAGE_PRESENT | PAGE_WRITABLE | PAGE_USER;
144        }
145
146        // map the kernel image at KERNEL_BASE -> kernel_phys, supervisor-only
147        // Note that KERNEL_BASE (0xffffffff80000000) lives at PML4 index 511
148        // and PDPT index 510 (the last 1GiB of the last 512GiB region).
149        unsafe extern "C" {
150            safe fn skernel();
151            safe fn ekernel();
152        }
153        let kernel_size = (ekernel as *const ()).addr() - (skernel as *const ()).addr();
154        let num_pages = (kernel_size + SZ_4K as usize - 1) / SZ_4K as usize;
155        let mut pts = [
156            &mut PT_KERNEL0,
157            &mut PT_KERNEL1,
158            &mut PT_KERNEL2,
159            &mut PT_KERNEL3,
160        ];
161        let num_pts = (num_pages + 511) / 512;
162        assert!(num_pts <= pts.len());
163        for (k, pt) in pts.iter_mut().enumerate().take(num_pts) {
164            for (j, entry) in pt.0.iter_mut().enumerate() {
165                let page = k * 512 + j;
166                if page < num_pages {
167                    *entry = (kernel_phys + page as u64 * SZ_4K) | PAGE_PRESENT | PAGE_WRITABLE;
168                }
169            }
170            PD_KERNEL.0[k] = phys_of(&**pt as *const _ as u64) | PAGE_PRESENT | PAGE_WRITABLE;
171        }
172        PDPT_KERNEL.0[510] = phys_of(&PD_KERNEL as *const _ as u64) | PAGE_PRESENT | PAGE_WRITABLE;
173
174        // build the top level
175        PML4.0[0] = phys_of(&PDPT_LOW as *const _ as u64)
176            | PAGE_PRESENT
177            | PAGE_WRITABLE
178            | PAGE_USER;
179        PML4.0[511] = phys_of(&PDPT_KERNEL as *const _ as u64) | PAGE_PRESENT | PAGE_WRITABLE;
180
181        // switch to the new page tables (also flushes the TLB)
182        let cr3 = phys_of(&PML4 as *const _ as u64);
183        asm!("mov cr3, {0}", in(reg) cr3, options(nostack));
184    }
185}
