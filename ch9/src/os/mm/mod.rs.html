23/// memory setup that must precede any heap allocation or `PerCpu` use: cache
24/// the Limine HHDM offset and initialize the kernel heap.
25///
26/// Since ch9 the console UART is a heap-allocated `Arc<dyn CharDevice>`
27/// (`drivers::UART`), so the heap must be ready before any `println!` (and
28/// before `cpu::init_cpu`, whose `info!` goes through the console). It is
29/// called once from `rust_main` before `init_cpu`; `init` must then not
30/// re-initialize the heap (that would drop existing allocations).
31pub fn init_early() {
32    // cache Limine's HHDM offset before any physical access
33    address::init_phys_virt_offset();
34    heap_allocator::init_heap();
35}
36
37/// initiate heap allocator, frame allocator and kernel space
38pub fn init() {
39    // `init_early` was already called (rust_main runs it before `init_cpu`);
40    // here we only finish the memory subsystem once GS (PerCpu) is up, which
41    // the interrupt-masking locks below require.
42    frame_allocator::init_frame_allocator();
43    // NX bits are ignored until EFER.NXE is set
44    crate::trap::msr::enable_nxe();
45    KERNEL_SPACE.exclusive_access().activate();
46}
