1//! VirtIO block device driver (VirtIO over PCI, `virtio-drivers` 0.2.0).

3//! The `Hal` implementation lives in [`crate::drivers::bus`].

10use virtio_drivers::transport::pci::PciTransport;

12/// How often the block request poll yields to the host. QEMU's single-threaded
13/// TCG only services the VirtIO device between translation blocks: a tight
14/// `spin_loop()` poll that never leaves its block can starve the device on SMP
15/// (the completion is never processed), so every so often we execute a
16/// privileged instruction that TCG emulates through a helper call, which
17/// forces the translator to exit the current block. On real hardware this is
18/// a harmless no-op for the poll.
19const POLL_YIELD_ITERS: u64 = 256;
20
21/// Yield the current translation block so QEMU's TCG can service the VirtIO
22/// device. `rdmsr` is always emulated by a helper call (it exits the block);
23/// `pause` would not. Reading the benign MSR 0x1B (APIC base) has no side
24/// effects.
25#[inline(never)]
26fn poll_yield() {
27    unsafe {
28        core::arch::asm!(
29            "rdmsr",
30            in("ecx") 0x1Bu32,
31            out("eax") _,
32            out("edx") _,
33            options(nostack)
34        );
35    }
36}

38/// Submit a VirtIO block request (read or write) and wait for completion,
39/// polling the used ring with an occasional `cpuid` yield (see `POLL_YIELD_ITERS`).
40fn blk_request(
41    blk: &mut VirtIOBlk<VirtioHal, PciTransport>,
42    block_id: usize,
43    is_write: bool,
44    buf: &mut [u8],
45) {
46    let mut req = BlkReq::default();
47    let mut resp = BlkResp::default();
48    let token = if is_write {
49        unsafe { blk.write_block_nb(block_id, &mut req, buf, &mut resp) }
50    } else {
51        unsafe { blk.read_block_nb(block_id, &mut req, buf, &mut resp) }
52    }
53    .expect("Error submitting VirtIOBlk request");
54    let mut spins: u64 = 0;
55    loop {
56        if blk.peek_used().is_some() {
57            let ret = if is_write {
58                unsafe { blk.complete_write_block(token, &req, buf, &mut resp) }
59            } else {
60                unsafe { blk.complete_read_block(token, &req, buf, &mut resp) }
61            };
62            ret.expect("Error completing VirtIOBlk request");
63            assert_eq!(resp.status(), RespStatus::OK, "VirtIOBlk request failed");
64            break;
65        }
66        spins += 1;
67        if spins % POLL_YIELD_ITERS == 0 {
68            // exit the translation block so the device completion is serviced
69            poll_yield();
70        }
71    }
72}

74pub struct VirtIOBlock(UPIntrFreeCell<VirtIOBlk<VirtioHal, PciTransport>>);
75
76// All access to the device goes through the `UPIntrFreeCell` in a single-threaded
77// kernel, so we can safely claim Send/Sync even though the raw pointers inside
78// `virtio-drivers`' PciTransport are not.
79unsafe impl Send for VirtIOBlock {}
80unsafe impl Sync for VirtIOBlock {}
81
82impl VirtIOBlock {
83    pub fn new() -> Self {
84        let mut pci = crate::drivers::pci::Pci::new();
85        let bdf = pci
86            .find_block()
87            .expect("no VirtIO block device found on the PCI bus");
88        let transport =
89            PciTransport::new::<VirtioHal>(pci.root(), bdf).expect("failed to init PCI transport");
90        let blk = VirtIOBlk::new(transport).expect("failed to init VirtIO block device");
91        info!("virtio-blk capacity: {} sectors", blk.capacity());
92        Self(unsafe { UPIntrFreeCell::new(blk) })
93    }
94}
95
96impl BlockDevice for VirtIOBlock {
97    fn read_block(&self, block_id: usize, buf: &mut [u8]) {
98        let mut blk = self.0.exclusive_access();
99        blk_request(&mut blk, block_id, false, buf);
100    }
101    fn write_block(&self, block_id: usize, buf: &[u8]) {
102        let mut blk = self.0.exclusive_access();
103        let mut wbuf = [0u8; 512];
104        wbuf.copy_from_slice(buf);
105        blk_request(&mut blk, block_id, true, &mut wbuf);
106    }
107}
