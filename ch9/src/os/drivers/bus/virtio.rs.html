1//! The `Hal` implementation used by every VirtIO device (ch9 `drivers/bus/virtio.rs`).
2//!
3//! On x86-64 the kernel maps all physical memory through a physmap at
4//! `HHDM` (Limine), so `phys_to_virt` is the physmap offset, DMA allocation
5//! uses the kernel frame allocator, and DMA addresses are the physical
6//! addresses translated through the kernel page table (`share`). This differs
7//! from the RISC-V ch9 `Hal`, which is an identity mapping and translates
8//! through `virt_to_phys`; the `virtio-drivers` 0.2.0 trait shape
9//! (`share`/`unshare`) is the one we keep.
10
11use crate::mm::{
12    FrameTracker, PhysAddr, PhysPageNum, StepByOne, VirtAddr, frame_alloc, frame_dealloc,
13    kernel_token, phys_to_virt,
14};
15use crate::sync::UPIntrFreeCell;
16use alloc::vec::Vec;
17use core::ptr::NonNull;
18use lazy_static::*;
19use virtio_drivers::{BufferDirection, Hal};
20
21/// The VirtIO HAL.
22pub struct VirtioHal;
23
24lazy_static! {
25    /// Frames backing the virtio DMA regions (kept alive until dealloc).
26    static ref QUEUE_FRAMES: UPIntrFreeCell<Vec<FrameTracker>> =
27        unsafe { UPIntrFreeCell::new(Vec::new()) };
28}
29
30impl Hal for VirtioHal {
31    fn dma_alloc(pages: usize) -> usize {
32        let mut ppn_base = PhysPageNum(0);
33        for i in 0..pages {
34            let frame = frame_alloc().unwrap();
35            if i == 0 {
36                ppn_base = frame.ppn;
37            }
38            // DMA needs contiguous physical pages; the stack allocator hands
39            // them out in order, so this holds in practice (same as the
40            // RISC-V tutorial).
41            assert_eq!(frame.ppn.0, ppn_base.0 + i);
42            QUEUE_FRAMES.exclusive_access().push(frame);
43        }
44        let pa: PhysAddr = ppn_base.into();
45        pa.0
46    }
47    fn dma_dealloc(pa: usize, pages: usize) -> i32 {
48        let pa = PhysAddr::from(pa);
49        let mut ppn_base: PhysPageNum = pa.into();
50        for _ in 0..pages {
51            frame_dealloc(ppn_base);
52            ppn_base.step();
53        }
54        0
55    }
56    fn phys_to_virt(paddr: usize) -> usize {
57        phys_to_virt(paddr)
58    }
59    fn share(buffer: NonNull<[u8]>, _direction: BufferDirection) -> usize {
60        let vaddr = buffer.as_ptr() as *mut u8 as usize;
61        crate::mm::PageTable::from_token(kernel_token())
62            .translate_va(VirtAddr::from(vaddr))
63            .unwrap()
64            .0
65    }
66    fn unshare(_paddr: usize, _buffer: NonNull<[u8]>, _direction: BufferDirection) {}
67}
