1//! Trap handling functionality
2//!
3//! For rCore on x86-64, there are two entry points:
4//!
5//! - the `syscall` entry, whose address is stored in the `LSTAR` MSR. It
6//!   synthesizes an interrupt-like frame and forwards to [`trap_handler()`].
7//! - the exception stubs, pointed to by the IDT. The CPU switches to the
8//!   current task's kernel stack (the `rsp0` field of the TSS) and pushes the
9//!   interrupt frame, then each stub forwards to [`trap_handler()`].
10//!
11//! [`trap_handler()`] then calls different functionality based on what
12//! exactly the exception was. For example, syscalls go to [`syscall()`], and
13//! the APIC timer interrupt (vector 32) triggers task preemption.
14
15// Boot-time CPU structures are installed once in [`init()`] and then never
16// touched again, so `static mut` (via raw pointers) is fine here.
17#![allow(static_mut_refs)]
18// `trap.S` is written in GNU/AT&T syntax.
19#![allow(bad_asm_style)]

45lazy_static! {
46    /// the shared IDT (the handler addresses are identical on every core; each
47    /// core `lidt`s it separately)
48    static ref IDT: InterruptDescriptorTable = {
49        let mut idt = InterruptDescriptorTable::new();
50        for i in 0..256 {
51            idt.set_handler(i, unsafe { exception_stub_addrs[i] } as u64);
52        }
53        idt
54    };
55}

79/// Switch the kernel stack that the CPU uses for traps to `top`.
80///
81/// Both the `rsp0` field of this core's TSS (used for user-mode exceptions)
82/// and its `current_stack_top` (used by `syscall_entry`) are updated. The task
83/// manager calls this right before switching to another thread.
84pub fn set_current_stack_top(top: u64) {
85    let pc = current_per_cpu_mut();
86    pc.current_stack_top = top as usize;
87    pc.tss.set_rsp0(top);
88}

91/// [`context::TrapContext::app_init_context`]) together with the APIC timer

93/// IF cleared (interrupt gates and `SFMASK`) so that external interrupts only
94/// happen in user mode, mirroring how RISC-V ch3 only enables `stimer` in
95/// user mode. Nothing needs to be done here.

98#[unsafe(no_mangle)]
99/// handle an interrupt, exception, or system call from user space
100pub fn trap_handler(cx: &mut TrapContext) -> &mut TrapContext {
101    match cx.trap_num {
102        TRAP_SYSCALL => {
103            let ret = syscall(cx.rax, [cx.rdi, cx.rsi, cx.rdx]);
104            // `cx` may have been replaced by `sys_exec` (it rebuilds the
105            // TrapContext on the kernel stack), so fetch it again
106            let cx = current_trap_cx();
107            cx.rax = ret as usize;
108        }
109        32 => {
110            // APIC timer interrupt (per-core local APIC)
111            timer::timer_eoi();
112            // only core 0 drives the global time base
113            if crate::cpu::current_cpu_id() == 0 {
114                timer::tick();
115            }
116            timer::set_next_trigger();
117            // wake up any threads whose sleep deadline has expired
118            timer::check_timer();
119            suspend_current_and_run_next();
120        }
121        v @ 33..=255 => {
122            // device IRQ while the CPU is in user mode: acknowledge it and
123            // dispatch to the registered handler, then return to the
124            // interrupted user code (no scheduling in IRQ context).
125            dispatch_irq(v);
126        }
127        14 => {
128            // page fault: CR2 holds the faulting linear address, error code
129            // bit 1 distinguishes a store (write) from a load (read) fault
130            let cr2: u64;
131            unsafe {
132                core::arch::asm!("mov rax, cr2", out("rax") cr2, options(nostack));
133            }
134            let access = if cx.error_code & 2 != 0 { "store" } else { "load" };
135            // diagnostic: identify the faulting thread and its user RSP
136            let (tid, rsp) = crate::task::current_task()
137                .map(|t| {
138                    let inner = t.inner_exclusive_access();
139                    (
140                        inner.res.as_ref().map(|r| r.tid).unwrap_or(usize::MAX),
141                        inner.get_trap_cx().rsp,
142                    )
143                })
144                .unwrap_or((usize::MAX, 0));
145            println!(
146                "[kernel] PageFault in application, {} addr = {:#x}, RIP = {:#x}, RSP = {:#x}, tid = {}, kernel killed it.",
147                access, cr2, cx.rip, rsp, tid
148            );
149            current_add_signal(SignalFlags::SIGSEGV);
150        }
151        13 | 6 => {
152            println!("[kernel] IllegalInstruction in application, kernel killed it.");
153            current_add_signal(SignalFlags::SIGILL);
154        }
155        _ => {
156            panic!(
157                "Unsupported trap: vector = {}, error_code = {:#x}, RIP = {:#x}",
158                cx.trap_num, cx.error_code, cx.rip
159            );
160        }
161    }
162    // handle the pending signals (including the ones just sent above)
163    handle_signals();
164    // the default action of an error signal is to kill the process
165    if let Some((errno, msg)) = check_signals_error_of_current() {
166        println!("[kernel] {}", msg);
167        exit_current_and_run_next(errno);
168    }
169    let cx = current_trap_cx();
170    cx
171}

173#[unsafe(no_mangle)]
174/// A trap that happened in kernel mode (i.e. a kernel bug): report and halt.
175pub extern "C" fn kernel_exception_panic(
176    vector: usize,
177    error_code: usize,
178    fault_rip: usize,
179    fault_cs: usize,
180) -> ! {
181    let cr2: u64;
182    let cr3: u64;
183    unsafe {
184        core::arch::asm!("mov rax, cr2", out("rax") cr2, options(nostack));
185        core::arch::asm!("mov rax, cr3", out("rax") cr3, options(nostack));
186    }
187    error!(
188        "[kernel] Exception #{} (error_code = {:#x}) occurred in kernel mode, CR2 = {:#x}, CR3 = {:#x}, RIP = {:#x}, CS = {:#x}, kernel halted.",
189        vector, error_code, cr2, cr3, fault_rip, fault_cs
190    );
191    crate::uart::shutdown(true)
192}

194#[unsafe(no_mangle)]
195/// The APIC timer interrupt while running in kernel mode (an idle core's
196/// `sti; hlt; cli` wakeup): EOI, advance the clock on core 0, and wake any
197/// threads whose sleep deadline has expired. Returns via `iretq` back to the
198/// idle loop.
199pub extern "C" fn kernel_timer_eoi() {
200    timer::timer_eoi();
201    if crate::cpu::current_cpu_id() == 0 {
202        timer::tick();
203    }
204    timer::check_timer();
205}

207/// Registered device IRQ handlers, indexed by IDT vector (`33..=255`).
208///
209/// Device IRQs (UART, virtio keyboard/mouse/gpu, ...) are delivered through
210/// the local APIC as IDT vectors above the exception range. A driver registers
211/// its handler here; `dispatch_irq` is invoked from both the user-mode
212/// `trap_handler` and the kernel-mode IRQ path (`kernel_irq_dispatch`).
213///
214/// Handlers run in interrupt context and must **never** schedule or block:
215/// they only drain device state and (at most) signal a condvar / enqueue a
216/// task. They may take locks (ch9 `UPIntrFreeCell`), which mask interrupts on
217/// the current core for the duration of the critical section.
218static mut IRQ_HANDLERS: [Option<fn()>; 256] = [None; 256];
219
220/// Register the handler for device IRQ `vector` (must be in `33..=255`).
221///
222/// Called once from `device_init` (boot, single-threaded).
223pub fn register_irq(vector: usize, handler: fn()) {
224    assert!(
225        (33..=255).contains(&vector),
226        "device IRQ vector {} out of range",
227        vector
228    );
229    unsafe {
230        IRQ_HANDLERS[vector] = Some(handler);
231    }
232    info!("irq: registered handler for vector {}", vector);
233}
234
235/// Acknowledge the interrupt (local APIC EOI) and run the registered handler
236/// for `vector`. Called from both the user-mode and kernel-mode trap paths.
237fn dispatch_irq(vector: usize) {
238    timer::timer_eoi();
239    let handler = unsafe { IRQ_HANDLERS[vector] };
240    match handler {
241        Some(h) => h(),
242        None => panic!("Unhandled device IRQ vector {}", vector),
243    }
244}

246#[unsafe(no_mangle)]
247/// A device IRQ that arrived while running in kernel mode. This currently
248/// happens only on an idle core's `sti; hlt` wakeup (the kernel otherwise
249/// keeps IF cleared); it acknowledges the interrupt and dispatches to the
250/// device handler, then returns via `iretq` to the interrupted kernel code.
251pub extern "C" fn kernel_irq_dispatch(vector: usize) {
252    dispatch_irq(vector);
253}
