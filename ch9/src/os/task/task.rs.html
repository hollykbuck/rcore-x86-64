1//! Implementation of [`TaskControlBlock`] (a thread).
2//!
3//! Since ch8 the scheduling unit is a **thread**; the process-wide state
4//! (address space, file descriptor table, signals, children, exit code) lives
5//! in [`ProcessControlBlock`](crate::task::ProcessControlBlock). Each thread
6//! owns its kernel stack (with the `TrapContext` on top, x86-64 style), its
7//! task context and its per-thread user stack.
8
9use super::id::TaskUserRes;
10use super::{KernelStack, ProcessControlBlock, TaskContext, kstack_alloc};
11use crate::trap::TrapContext;
12use crate::{sync::UPIntrFreeCell};
13use alloc::sync::{Arc, Weak};
14use core::sync::atomic::AtomicUsize;
15
16/// A thread of a process (the scheduling unit since ch8).
17pub struct TaskControlBlock {
18    /// the process this thread belongs to
19    pub process: Weak<ProcessControlBlock>,
20    /// the kernel stack of this thread
21    pub kstack: KernelStack,
22    /// the id of the processor that last ran (or will run) this thread. It is
23    /// the owner of this thread's context save: a woken thread must be parked
24    /// on its owner's pending queue and only re-enter the global ready queue
25    /// after the owner has switched away from it (SMP race, see ch8 stage 3).
26    pub last_cpu: AtomicUsize,
27    // mutable
28    inner: UPIntrFreeCell<TaskControlBlockInner>,
29}
30
31impl TaskControlBlock {
32    /// Get the mutable reference to the inner structure of the thread.
33    pub fn inner_exclusive_access(&self) -> crate::sync::UPIntrRefMut<'_, TaskControlBlockInner> {
34        self.inner.exclusive_access()
35    }
36
37    /// Get the token of the address space of the process this thread belongs to.
38    pub fn get_user_token(&self) -> usize {
39        let process = self.process.upgrade().unwrap();
40        let inner = process.inner_exclusive_access();
41        inner.memory_set.token()
42    }
43}
44
45pub struct TaskControlBlockInner {
46    /// per-thread user resources (tid and user stack)
47    pub res: Option<TaskUserRes>,
48    /// virtual address of the `TrapContext` on this thread's kernel stack
49    pub trap_cx_ptr: usize,
50    /// the context to resume this thread's kernel flow after scheduling
51    pub task_cx: TaskContext,
52    /// the scheduling state of this thread
53    pub task_status: TaskStatus,
54    /// the exit code, set when the thread exits
55    pub exit_code: Option<i32>,
56}
57
58impl TaskControlBlockInner {
59    pub fn get_trap_cx(&self) -> &'static mut TrapContext {
60        unsafe { &mut *(self.trap_cx_ptr as *mut TrapContext) }
61    }
62}
63
64impl TaskControlBlock {
65    /// Create a thread of `process`. If `alloc_user_res` is set, a new user
66    /// stack is mapped for it (used by `sys_thread_create`); the main thread
67    /// of a freshly loaded/forked process has its stack already mapped by
68    /// `MemorySet::from_elf`/`from_existed_user`, so it passes `false`.
69    pub fn new(
70        process: Arc<ProcessControlBlock>,
71        ustack_base: usize,
72        alloc_user_res: bool,
73    ) -> Self {
74        let res = TaskUserRes::new(Arc::clone(&process), ustack_base, alloc_user_res);
75        let kstack = kstack_alloc();
76        // the trap context is written by the caller afterwards; reserve the
77        // slot at the top of the kernel stack now so `trap_cx_ptr` is stable
78        let trap_cx_ptr = kstack.push_context(TrapContext::app_init_context(0, 0));
79        Self {
80            process: Arc::downgrade(&process),
81            kstack,
82            last_cpu: AtomicUsize::new(0),
83            inner: unsafe {
84                UPIntrFreeCell::new(TaskControlBlockInner {
85                    res: Some(res),
86                    trap_cx_ptr,
87                    task_cx: TaskContext::goto_restore(trap_cx_ptr),
88                    task_status: TaskStatus::Ready,
89                    exit_code: None,
90                })
91            },
92        }
93    }
94}

96/// The scheduling state of a thread.
97#[derive(Copy, Clone, PartialEq)]
98pub enum TaskStatus {
99    /// waiting in the ready queue
100    Ready,
101    /// running on a processor
102    Running,
103    /// blocked (waiting for a timer or a synchronization primitive)
104    Blocked,
105}
